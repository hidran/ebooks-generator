# Lezione 105. Configurazione centralizzata e service discovery

## Punti chiave

### Configurazione sparsa, indirizzi fissi

- Le stesse righe di log copiate in ogni application.yml
- Gli ordini conoscono la porta del catalogo: 127.0.0.1:21050
- Un cambiamento richiede modifica e riavvio di ogni servizio
- DRY violato anche nella configurazione

### Config Server ed Eureka

- Config Server (21054): file in config-repo, profilo native
- application.yml per tutti, order-service.yml solo per gli ordini
- spring.config.import: ogni servizio chiede la sua configurazione
- Eureka (21055): ogni servizio si registra con il suo nome
- lb://catalog-service: il load balancer risolve il nome nel registro

> Percorso: Servizio, poi Config Server, poi Eureka, poi Catalogo.

### Quando non servono (KISS)

- Kubernetes: Service e DNS rendono superfluo Eureka
- ConfigMap e Secret possono sostituire il Config Server
- Pochi servizi stabili: bastano le variabili d'ambiente
- Segreti mai nei file di configurazione condivisi

`config-server/src/test/java/com/shop/config/ConfigServerTests.java`

~~~java
package com.shop.config;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import org.junit.jupiter.api.Test;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.web.server.LocalServerPort;
import static org.assertj.core.api.Assertions.assertThat;

@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT,
        properties = "spring.cloud.config.server.native.search-locations=file:../config-repo")
class ConfigServerTests {
    @LocalServerPort
    int port;

    @Test
    void everyServiceGetsTheSharedSettings() throws Exception {
        String body = get("/catalog-service/default");
        assertThat(body).contains("\"eureka.client.service-url.defaultZone\":\"http://127.0.0.1:21055/eureka/\"");
        assertThat(body).contains("\"logging.level.root\":\"warn\"");
    }

    @Test
    void orderServiceFindsTheCatalogByName() throws Exception {
        String body = get("/order-service/default");
        assertThat(body).contains("\"spring.http.serviceclient.catalog.base-url\":\"lb://catalog-service\"");
        assertThat(body).contains("\"eureka.client.service-url.defaultZone\"");
    }

    private String get(String path) throws Exception {
        HttpRequest request = HttpRequest.newBuilder(URI.create("http://127.0.0.1:" + port + path)).build();
        return HttpClient.newHttpClient().send(request, HttpResponse.BodyHandlers.ofString()).body();
    }
}
~~~

~~~java
package com.shop.orders;

import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.io.UncheckedIOException;
import java.net.InetSocketAddress;
import java.nio.charset.StandardCharsets;
import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.test.context.DynamicPropertyRegistry;
import org.springframework.test.context.DynamicPropertySource;
import org.springframework.test.web.servlet.MockMvc;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest(properties = {"spring.cloud.config.enabled=false", "eureka.client.enabled=false",
        "spring.http.serviceclient.catalog.base-url=lb://catalog-service"})
@AutoConfigureMockMvc
class OrderApiTests {
    static final HttpServer CATALOG = catalogStub();

    @Autowired
    MockMvc mvc;

    @DynamicPropertySource
    static void discovery(DynamicPropertyRegistry registry) {
        registry.add("spring.cloud.discovery.client.simple.instances.catalog-service[0].uri",
                () -> "http://127.0.0.1:" + CATALOG.getAddress().getPort());
    }

    @AfterAll
    static void stopStub() {
        CATALOG.stop(0);
    }

    @Test
    void findsTheCatalogByName() throws Exception {
        mvc.perform(post("/api/orders").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"productId\":1,\"quantity\":2}"))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.totalInCents").value(2468));
    }

    @Test
    void unknownOrderIsNotFound() throws Exception {
        mvc.perform(get("/api/orders/99")).andExpect(status().isNotFound());
    }

    private static HttpServer catalogStub() {
        try {
            HttpServer server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
            server.createContext("/api/products/1", exchange -> {
                byte[] body = "{\"id\":1,\"name\":\"Stub\",\"priceInCents\":1234}".getBytes(StandardCharsets.UTF_8);
                exchange.getResponseHeaders().add("Content-Type", "application/json");
                exchange.sendResponseHeaders(200, body.length);
                exchange.getResponseBody().write(body);
                exchange.close();
            });
            server.start();
            return server;
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }
}
~~~

## Riepilogo

- Impostazioni comuni in un solo file: DRY
- Config Server: application.yml e order-service.yml
- Eureka registra catalogo e ordini per nome
- Gli ordini chiamano lb://catalog-service, non una porta
- Avvio ordinato: registro e configurazione per primi
