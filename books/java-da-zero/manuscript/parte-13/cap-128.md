# Lezione 103. Il pattern outbox

## Punti chiave

### La doppia scrittura

- Salvare l'ordine nel database
- Pubblicare l'evento su Kafka
- Due sistemi, nessuna transazione comune
- Kafka fermo: ordine rifiutato o evento perso

### Il pattern outbox


> Percorso: OrderService, poi orders + outbox, poi OutboxRelay, poi Kafka.

- L'evento è una riga nella stessa transazione
- Un relay pianificato legge le righe in attesa
- Pubblica, poi segna sent_at
- Almeno una volta: consumatori idempotenti

### Che cosa non serve

- Transazioni distribuite tra database e Kafka
- Change data capture per un solo servizio
- Una libreria: bastano una tabella e un metodo pianificato

`order-service/src/main/java/com/shop/orders/OrderService.java`

~~~java
package com.shop.orders;

import java.util.Optional;
import org.springframework.stereotype.Service;
import org.springframework.transaction.support.TransactionTemplate;

@Service
public class OrderService {
    private final ProductLookup products;
    private final OrderRepository orders;
    private final OrderEventPublisher events;
    private final TransactionTemplate transaction;

    public OrderService(ProductLookup products, OrderRepository orders, OrderEventPublisher events,
            TransactionTemplate transaction) {
        this.products = products;
        this.orders = orders;
        this.events = events;
        this.transaction = transaction;
    }

    public Order place(long productId, int quantity) {
        ProductSnapshot product = products.find(productId)
                .orElseThrow(() -> new UnknownProductException(productId));
        return transaction.execute(status -> {
            Order order = orders.save(product.productId(), product.name(), product.priceInCents(),
                    quantity);
            events.publish(OrderPlaced.from(order));
            return order;
        });
    }

    public Optional<Order> find(long id) {
        return orders.findById(id);
    }
}
~~~

`order-service/src/main/java/com/shop/orders/OutboxRelay.java`

~~~java
package com.shop.orders;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

@Component
public class OutboxRelay {
    private static final Logger log = LoggerFactory.getLogger(OutboxRelay.class);

    private final Outbox outbox;
    private final MessageSender sender;

    public OutboxRelay(Outbox outbox, MessageSender sender) {
        this.outbox = outbox;
        this.sender = sender;
    }

    @Scheduled(fixedDelayString = "${outbox.relay.delay}",
            initialDelayString = "${outbox.relay.delay}")
    public void publishPending() {
        for (OutboxMessage message : outbox.pending(10)) {
            try {
                sender.send(message.topic(), message.messageKey(), message.payload());
            } catch (RuntimeException exception) {
                log.warn("Kafka unavailable: outbox message {} stays pending", message.id());
                return;
            }
            outbox.markSent(message.id());
            log.info("Outbox message {} sent to {}", message.id(), message.topic());
        }
    }
}
~~~

`order-service/src/main/java/com/shop/orders/OrderRepository.java`

~~~java
package com.shop.orders;

import java.util.Optional;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.jdbc.support.GeneratedKeyHolder;
import org.springframework.jdbc.support.KeyHolder;
import org.springframework.stereotype.Repository;

@Repository
public class OrderRepository {
    private final JdbcClient jdbc;

    public OrderRepository(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    public Order save(long productId, String productName, int unitPriceInCents, int quantity) {
        KeyHolder keys = new GeneratedKeyHolder();
        jdbc.sql("""
                insert into orders (product_id, product_name, unit_price_in_cents, quantity)
                values (?, ?, ?, ?)""")
                .params(productId, productName, unitPriceInCents, quantity)
                .update(keys, "id");
        long id = keys.getKey().longValue();
        return new Order(id, productId, productName, unitPriceInCents, quantity);
    }

    public Optional<Order> findById(long id) {
        return jdbc.sql("""
                select id, product_id, product_name, unit_price_in_cents, quantity
                from orders where id = ?""")
                .param(id)
                .query(Order.class)
                .optional();
    }
}
~~~

`order-service/src/main/java/com/shop/orders/Outbox.java`

~~~java
package com.shop.orders;

import java.util.List;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Repository;

@Repository
public class Outbox {
    private final JdbcClient jdbc;

    public Outbox(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    public void add(String topic, String key, String payload) {
        jdbc.sql("""
                insert into outbox (topic, message_key, payload, created_at)
                values (?, ?, ?, current_timestamp)""")
                .params(topic, key, payload)
                .update();
    }

    public List<OutboxMessage> pending(int limit) {
        return jdbc.sql("""
                select id, topic, message_key, payload from outbox
                where sent_at is null order by id limit ?""")
                .param(limit)
                .query(OutboxMessage.class)
                .list();
    }

    public void markSent(long id) {
        jdbc.sql("update outbox set sent_at = current_timestamp where id = ?")
                .param(id)
                .update();
    }
}
~~~

~~~java
    }

    @Test
    void reportsAProductTheCatalogDoesNotKnow() throws Exception {
        placeOrder(99).andExpect(status().isNotFound());
    }

        delayMillis = 2000;
        placeOrder(1).andExpect(status().isServiceUnavailable())
                .andExpect(jsonPath("$.type").value("https://api.shop.example/problems/catalog-unavailable"));
    }

    @Test
    void retriesATemporaryFailure() throws Exception {
        failuresLeft = 1;
        placeOrder(1).andExpect(status().isCreated());
        assertThat(calls.get()).isEqualTo(2);
    }

        return mvc.perform(post("/api/orders").contentType(MediaType.APPLICATION_JSON)
                .content("{\"productId\":" + productId + ",\"quantity\":2}"));
    }

        calls.incrementAndGet();
        try {
            Thread.sleep(delayMillis);
        } catch (InterruptedException exception) {
            Thread.currentThread().interrupt();
        }
        if (failuresLeft > 0) {
            failuresLeft--;
            exchange.sendResponseHeaders(503, -1);
        } else if (exchange.getRequestURI().getPath().endsWith("/1")) {
            byte[] body = "{\"id\":1,\"name\":\"Keyboard\",\"priceInCents\":4900}".getBytes(StandardCharsets.UTF_8);
            exchange.getResponseHeaders().add("Content-Type", "application/json");
            exchange.sendResponseHeaders(200, body.length);
            exchange.getResponseBody().write(body);
        } else {
            exchange.sendResponseHeaders(404, -1);
        }
        exchange.close();
    }
}
~~~

~~~java
class OrderEventsTests {
    @Autowired
    OrderService orders;

    @Autowired
    OutboxRelay relay;

    @Autowired
    EmbeddedKafkaBroker broker;

    @Autowired
    JsonMapper json;

    @MockitoBean
    ProductLookup products;

    @Test
    void theRelayPublishesOrderPlacedAsJson() {
        given(products.find(1L)).willReturn(Optional.of(new ProductSnapshot(1L, "Keyboard", 4900)));
        Order order = orders.place(1L, 2);
        relay.publishPending();
        Map<String, Object> props = KafkaTestUtils.consumerProps(broker, "test", true);
        props.put("auto.offset.reset", "earliest");
        try (Consumer<String, String> consumer = new DefaultKafkaConsumerFactory<>(props,
                new StringDeserializer(), new StringDeserializer()).createConsumer()) {
            broker.consumeFromAnEmbeddedTopic(consumer, "orders.placed");
            ConsumerRecord<String, String> record = KafkaTestUtils.getSingleRecord(consumer, "orders.placed",
                    Duration.ofSeconds(10));
            JsonNode event = json.readTree(record.value());
            assertThat(record.key()).isEqualTo(String.valueOf(order.id()));
            assertThat(event.get("orderId").asLong()).isEqualTo(order.id());
            assertThat(event.get("productId").asLong()).isEqualTo(1L);
            assertThat(event.get("quantity").asInt()).isEqualTo(2);
            assertThat(event.get("totalInCents").asInt()).isEqualTo(9800);
        }
    }
}
~~~

~~~java
package com.shop.orders;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.ArgumentMatchers.anyString;
import static org.mockito.ArgumentMatchers.contains;
import static org.mockito.ArgumentMatchers.eq;
import static org.mockito.BDDMockito.given;
import static org.mockito.BDDMockito.willThrow;
import static org.mockito.Mockito.times;
import static org.mockito.Mockito.verify;
import static org.mockito.Mockito.verifyNoInteractions;

import java.util.Optional;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.test.context.bean.override.mockito.MockitoBean;

@SpringBootTest
class OutboxTests {
    @Autowired
    OrderService orders;

    @Autowired
    OutboxRelay relay;

    @Autowired
    Outbox outbox;

    @Autowired
    JdbcClient jdbc;

    @MockitoBean
    ProductLookup products;

    @MockitoBean
    MessageSender sender;

    @BeforeEach
    void emptyTablesAndAKnownProduct() {
        jdbc.sql("delete from outbox").update();
        jdbc.sql("delete from orders").update();
        given(products.find(1L)).willReturn(Optional.of(new ProductSnapshot(1L, "Keyboard", 4900)));
    }

    @Test
    void savesTheEventInTheSameTransactionAsTheOrder() {
        Order order = orders.place(1L, 2);
        assertThat(outbox.pending(10)).singleElement().satisfies(message -> {
            assertThat(message.topic()).isEqualTo("orders.placed");
            assertThat(message.messageKey()).isEqualTo(String.valueOf(order.id()));
            assertThat(message.payload()).contains("\"totalInCents\":9800");
        });
        verifyNoInteractions(sender);
    }

    @Test
    void keepsTheEventWhileKafkaIsDown() {
        willThrow(new IllegalStateException("broker down")).willDoNothing()
                .given(sender).send(anyString(), anyString(), anyString());
        Order order = orders.place(1L, 2);
        relay.publishPending();
        assertThat(outbox.pending(10)).hasSize(1);
        relay.publishPending();
        assertThat(outbox.pending(10)).isEmpty();
        verify(sender, times(2)).send(eq("orders.placed"), eq(String.valueOf(order.id())),
                contains("\"orderId\":" + order.id()));
    }

    @Test
    void sendsEachEventOnlyOnce() {
        orders.place(1L, 2);
        relay.publishPending();
        relay.publishPending();
        verify(sender, times(1)).send(anyString(), anyString(), anyString());
    }
}
~~~

## Riepilogo

- Ordine ed evento nella stessa transazione
- OrderService non cambia: cambia l'adattatore
- Il relay pubblica e segna le righe inviate
- Kafka fermo: ordine accettato, evento in attesa
- Kafka di nuovo attivo: nessun evento perso
