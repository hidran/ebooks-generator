# Lezione 102. Messaggi asincroni con Kafka

## Punti chiave

### Comunicare con eventi

- Gli ordini pubblicano un fatto: OrderPlaced
- Chi è interessato si iscrive
- Il produttore non conosce i consumatori
- Notifiche ferme? Gli eventi aspettano nel topic

### Kafka in breve


> Percorso: order-service, poi orders.placed, poi notification-service.

- Topic: registro ordinato e durevole
- Chiave orderId: stesso ordine, stessa partizione
- Gruppo di consumatori: notification-service

### Il contratto è JSON

- OrderPlaced(orderId, productId, quantity, totalInCents)
- Ogni servizio ha il suo record
- Nessun header con il tipo Java
- Non servono oggi: Schema Registry, Avro, Kafka Streams

`order-service/src/main/java/com/shop/orders/KafkaOrderEventPublisher.java`

~~~java
package com.shop.orders;

import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Component;

@Component
public class KafkaOrderEventPublisher implements OrderEventPublisher {
    static final String TOPIC = "orders.placed";

    private final KafkaTemplate<String, OrderPlaced> kafka;

    public KafkaOrderEventPublisher(KafkaTemplate<String, OrderPlaced> kafka) {
        this.kafka = kafka;
    }

    @Override
    public void publish(OrderPlaced event) {
        kafka.send(TOPIC, String.valueOf(event.orderId()), event);
    }
}
~~~

~~~java
package com.shop.notifications;

import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.stereotype.Component;

@Component
public class OrderPlacedListener {
    private final NotificationSender sender;

    public OrderPlacedListener(NotificationSender sender) {
        this.sender = sender;
    }

    @KafkaListener(topics = "orders.placed")
    public void on(OrderPlaced event) {
        sender.send("order " + event.orderId() + " placed, total "
                + event.totalInCents() + " cents");
    }
}
~~~

`order-service/src/main/java/com/shop/orders/OrderService.java`

~~~java
package com.shop.orders;

import java.util.Optional;
import org.springframework.stereotype.Service;

@Service
public class OrderService {
    private final ProductLookup products;
    private final OrderRepository orders;
    private final OrderEventPublisher events;

    public OrderService(ProductLookup products, OrderRepository orders, OrderEventPublisher events) {
        this.products = products;
        this.orders = orders;
        this.events = events;
    }

    public Order place(long productId, int quantity) {
        ProductSnapshot product = products.find(productId)
                .orElseThrow(() -> new UnknownProductException(productId));
        Order order = new Order(orders.nextId(), product.productId(), product.name(),
                product.priceInCents(), quantity);
        orders.save(order);
        events.publish(OrderPlaced.from(order));
        return order;
    }

    public Optional<Order> find(long id) {
        return orders.findById(id);
    }
}
~~~

~~~java
        catalog.setExecutor(Executors.newCachedThreadPool());
        catalog.createContext("/api/products/", OrderApiTests::answer);

        catalog.stop(0);
    }

    @BeforeEach
    void healthyCatalog() {
        calls.set(0);
        failuresLeft = 0;
        delayMillis = 0;
    }

    @Test
    void copiesNameAndPriceFromTheCatalog() throws Exception {
        placeOrder(1).andExpect(status().isCreated())
                .andExpect(jsonPath("$.productName").value("Keyboard"))
                .andExpect(jsonPath("$.totalInCents").value(9800));
    }

    @Test
    void reportsAProductTheCatalogDoesNotKnow() throws Exception {
        placeOrder(99).andExpect(status().isNotFound());
    }

    @Test
    void answersQuicklyWhenTheCatalogIsTooSlow() throws Exception {
        delayMillis = 2000;
        placeOrder(1).andExpect(status().isServiceUnavailable())
                .andExpect(jsonPath("$.type").value("https://api.shop.example/problems/catalog-unavailable"));
    }

    @Test
    void retriesATemporaryFailure() throws Exception {
        failuresLeft = 1;
        placeOrder(1).andExpect(status().isCreated());
        assertThat(calls.get()).isEqualTo(2);
    }

    private ResultActions placeOrder(long productId) throws Exception {
        return mvc.perform(post("/api/orders").contentType(MediaType.APPLICATION_JSON)
                .content("{\"productId\":" + productId + ",\"quantity\":2}"));
    }

    private static void answer(HttpExchange exchange) throws IOException {
        calls.incrementAndGet();
        try {
            Thread.sleep(delayMillis);
        } catch (InterruptedException exception) {
            Thread.currentThread().interrupt();
        }
        if (failuresLeft > 0) {
            failuresLeft--;
            exchange.sendResponseHeaders(503, -1);
        } else if (exchange.getRequestURI().getPath().endsWith("/1")) {
            byte[] body = "{\"id\":1,\"name\":\"Keyboard\",\"priceInCents\":4900}".getBytes(StandardCharsets.UTF_8);
            exchange.getResponseHeaders().add("Content-Type", "application/json");
            exchange.sendResponseHeaders(200, body.length);
            exchange.getResponseBody().write(body);
        } else {
            exchange.sendResponseHeaders(404, -1);
        }
        exchange.close();
    }
}
~~~

~~~java
package com.shop.orders;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.BDDMockito.given;

import java.time.Duration;
import java.util.Map;
import java.util.Optional;
import org.apache.kafka.clients.consumer.Consumer;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.common.serialization.StringDeserializer;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.kafka.core.DefaultKafkaConsumerFactory;
import org.springframework.kafka.test.EmbeddedKafkaBroker;
import org.springframework.kafka.test.context.EmbeddedKafka;
import org.springframework.kafka.test.utils.KafkaTestUtils;
import org.springframework.test.context.bean.override.mockito.MockitoBean;
import tools.jackson.databind.JsonNode;
import tools.jackson.databind.json.JsonMapper;

@SpringBootTest(properties = "spring.kafka.bootstrap-servers=${spring.embedded.kafka.brokers}")
@EmbeddedKafka(topics = "orders.placed")
class OrderEventsTests {
    @Autowired
    OrderService orders;

    @Autowired
    EmbeddedKafkaBroker broker;

    @Autowired
    JsonMapper json;

    @MockitoBean
    ProductLookup products;

    @Test
    void placingAnOrderPublishesOrderPlacedAsJson() {
        given(products.find(1L)).willReturn(Optional.of(new ProductSnapshot(1L, "Keyboard", 4900)));
        Order order = orders.place(1L, 2);
        Map<String, Object> props = KafkaTestUtils.consumerProps(broker, "test", true);
        props.put("auto.offset.reset", "earliest");
        try (Consumer<String, String> consumer = new DefaultKafkaConsumerFactory<>(props,
                new StringDeserializer(), new StringDeserializer()).createConsumer()) {
            broker.consumeFromAnEmbeddedTopic(consumer, "orders.placed");
            ConsumerRecord<String, String> record = KafkaTestUtils.getSingleRecord(consumer, "orders.placed",
                    Duration.ofSeconds(10));
            JsonNode event = json.readTree(record.value());
            assertThat(record.key()).isEqualTo(String.valueOf(order.id()));
            assertThat(event.get("orderId").asLong()).isEqualTo(order.id());
            assertThat(event.get("quantity").asInt()).isEqualTo(2);
            assertThat(event.get("totalInCents").asInt()).isEqualTo(9800);
            assertThat(record.headers().lastHeader("__TypeId__")).isNull();
        }
    }
}
~~~

## Riepilogo

- Evento: un fatto pubblicato, senza attendere risposte
- KafkaTemplate pubblica, @KafkaListener consuma
- JSON come contratto, un record per servizio
- Test con il broker incorporato, senza Docker
- Un solo broker KRaft in Docker per la prova reale
