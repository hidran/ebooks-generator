# Lezione 107. Coerenza distribuita con le saghe

## Punti chiave

### Un ordine, due archivi

- Il catalogo riserva la merce, gli ordini incassano
- Nessuna transazione unica tra due servizi
- Pagamento rifiutato dopo la riserva: magazzino bloccato
- Serve una regola per annullare i passi già fatti

### Saga: coreografia o orchestrazione

- Saga: transazioni locali collegate da eventi
- Compensazione: un'azione che annulla un passo
- Coreografia: ogni servizio reagisce agli eventi degli altri
- Orchestrazione: un coordinatore dice a ciascuno che cosa fare
- Due partecipanti: coreografia (KISS)

### Il nostro flusso

- orders.placed: ordine PENDING, merce da riservare
- stock.reserved oppure stock.rejected dal catalogo
- Pagamento accettato: CONFIRMED
- Pagamento rifiutato: orders.cancelled, merce rilasciata

> Percorso: Ordine creato, poi Merce riservata, poi Pagamento, poi Confermato o compensato.

### Quando non serve (KISS)

- Un solo servizio: basta una transazione locale
- Pochi passi: coreografia; molti passi: orchestratore
- In produzione: eventi pubblicati con l'outbox
- Consistenza finale: stati intermedi visibili (PENDING)

~~~java
package com.shop.orders;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Component;

@Component
public class OrderSaga {
    private static final Logger log = LoggerFactory.getLogger(OrderSaga.class);
    private final OrderService orders;
    private final OrderRepository repository;
    private final PaymentGateway payments;
    private final KafkaTemplate<String, Object> events;

    public OrderSaga(OrderService orders, OrderRepository repository, PaymentGateway payments,
            KafkaTemplate<String, Object> events) {
        this.orders = orders;
        this.repository = repository;
        this.payments = payments;
        this.events = events;
    }

    @KafkaListener(topics = "stock.rejected")
    public void onStockRejected(StockRejected event) {
        repository.save(orders.find(event.orderId()).withStatus(OrderStatus.CANCELLED));
        log.info("Order {} cancelled: stock not available", event.orderId());
    }

    @KafkaListener(topics = "stock.reserved")
    public void onStockReserved(StockReserved event) {
        Order order = orders.find(event.orderId());
        if (payments.charge(order.id(), order.totalInCents())) {
            repository.save(order.withStatus(OrderStatus.CONFIRMED));
            log.info("Order {} confirmed: payment accepted", order.id());
            return;
        }
        repository.save(order.withStatus(OrderStatus.CANCELLED));
        events.send("orders.cancelled", String.valueOf(order.id()),
                new OrderCancelled(order.id(), order.productId(), order.quantity()));
        log.info("Order {} cancelled: payment refused", order.id());
    }
}
~~~

`catalog-service/src/main/java/com/shop/catalog/StockListener.java`

~~~java
package com.shop.catalog;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Component;

@Component
public class StockListener {
    private static final Logger log = LoggerFactory.getLogger(StockListener.class);
    private final StockService stock;
    private final KafkaTemplate<String, Object> events;

    public StockListener(StockService stock, KafkaTemplate<String, Object> events) {
        this.stock = stock;
        this.events = events;
    }

    @KafkaListener(topics = "orders.placed")
    public void onOrderPlaced(OrderPlaced order) {
        String key = String.valueOf(order.orderId());
        if (stock.reserve(order.productId(), order.quantity())) {
            log.info("Stock reserved for order {}", order.orderId());
            events.send("stock.reserved", key, new StockReserved(order.orderId()));
        } else {
            log.info("Stock rejected for order {}", order.orderId());
            events.send("stock.rejected", key, new StockRejected(order.orderId()));
        }
    }
    @KafkaListener(topics = "orders.cancelled")
    public void onOrderCancelled(OrderCancelled order) {
        stock.release(order.productId(), order.quantity());
        log.info("Stock released for order {}", order.orderId());
    }
}
~~~

~~~java
package com.shop.catalog;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import org.springframework.stereotype.Service;

@Service
public class StockService {
    private final Map<Long, Integer> available = new ConcurrentHashMap<>(Map.of(1L, 10, 2L, 5, 3L, 3));

    public synchronized boolean reserve(long productId, int quantity) {
        int stock = available.getOrDefault(productId, 0);
        if (stock < quantity) {
            return false;
        }
        available.put(productId, stock - quantity);
        return true;
    }

    public synchronized void release(long productId, int quantity) {
        available.merge(productId, quantity, Integer::sum);
    }

    public int available(long productId) {
        return available.getOrDefault(productId, 0);
    }
}
~~~

~~~java
package com.shop.catalog;

import java.time.Duration;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.context.TestConfiguration;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.test.context.EmbeddedKafka;
import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;

@SpringBootTest(properties = "spring.kafka.bootstrap-servers=${spring.embedded.kafka.brokers}")
@EmbeddedKafka(partitions = 1, topics = {"orders.placed", "orders.cancelled", "stock.reserved", "stock.rejected"})
class StockSagaTests {
    @Autowired
    KafkaTemplate<String, Object> kafka;

    @Autowired
    StockService stock;

    @Autowired
    StockEvents published;

    @Test
    void reservesTheStockOfAPlacedOrder() {
        kafka.send("orders.placed", "1", new OrderPlaced(1, 1, 2, 9800));
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() -> {
            assertThat(stock.available(1)).isEqualTo(8);
            assertThat(published.events).contains("stock.reserved {\"orderId\":1}");
        });
    }

    @Test
    void rejectsAnOrderWithoutStock() {
        kafka.send("orders.placed", "2", new OrderPlaced(2, 2, 50, 125000));
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() ->
                assertThat(published.events).contains("stock.rejected {\"orderId\":2}"));
        assertThat(stock.available(2)).isEqualTo(5);
    }

    @Test
    void releasesTheStockOfACancelledOrder() {
        kafka.send("orders.placed", "3", new OrderPlaced(3, 3, 2, 37800));
        await().atMost(Duration.ofSeconds(10)).until(() -> stock.available(3) == 1);
        kafka.send("orders.cancelled", "3", new OrderCancelled(3, 3, 2));
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() -> assertThat(stock.available(3)).isEqualTo(3));
    }

    @TestConfiguration
    static class StockEvents {
        final List<String> events = new CopyOnWriteArrayList<>();

        @KafkaListener(topics = {"stock.reserved", "stock.rejected"}, groupId = "catalog-test")
        void capture(ConsumerRecord<String, String> record) {
            events.add(record.topic() + " " + record.value());
        }
    }
}
~~~

~~~java
package com.shop.orders;

import java.util.Map;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;
import org.springframework.stereotype.Repository;

@Repository
public class OrderRepository {
    private final Map<Long, Order> orders = new ConcurrentHashMap<>();
    private final AtomicLong nextId = new AtomicLong(1);

    public Order create(long productId, int quantity, int totalInCents) {
        Order order = new Order(nextId.getAndIncrement(), productId, quantity, totalInCents, OrderStatus.PENDING);
        orders.put(order.id(), order);
        return order;
    }

    public void save(Order order) {
        orders.put(order.id(), order);
    }

    public Optional<Order> findById(long id) {
        return Optional.ofNullable(orders.get(id));
    }
}
~~~

~~~java
package com.shop.orders;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;

@Service
public class OrderService {
    private static final Logger log = LoggerFactory.getLogger(OrderService.class);
    private final CatalogClient catalog;
    private final OrderRepository orders;
    private final KafkaTemplate<String, Object> events;

    public OrderService(CatalogClient catalog, OrderRepository orders, KafkaTemplate<String, Object> events) {
        this.catalog = catalog;
        this.orders = orders;
        this.events = events;
    }

    public Order place(long productId, int quantity) {
        ProductView product = catalog.product(productId);
        Order order = orders.create(productId, quantity, product.priceInCents() * quantity);
        events.send("orders.placed", String.valueOf(order.id()),
                new OrderPlaced(order.id(), productId, quantity, order.totalInCents()));
        log.info("Order {} placed: waiting for stock", order.id());
        return order;
    }

    public Order find(long id) {
        return orders.findById(id).orElseThrow(() -> new OrderNotFoundException(id));
    }
}
~~~

~~~java
import java.time.Duration;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.context.TestConfiguration;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;

import org.springframework.test.context.bean.override.mockito.MockitoBean;
import org.springframework.test.web.servlet.MockMvc;
import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;
import static org.mockito.BDDMockito.given;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest(properties = "spring.kafka.bootstrap-servers=${spring.embedded.kafka.brokers}")
@EmbeddedKafka(partitions = 1, topics = {"orders.placed", "orders.cancelled", "stock.reserved", "stock.rejected"})
@AutoConfigureMockMvc
class OrderSagaTests {
    @Autowired
    MockMvc mvc;

    @Autowired
    KafkaTemplate<String, Object> kafka;

    @Autowired
    OrderRepository orders;

    @Autowired
    OrderEvents published;

    @MockitoBean
    CatalogClient catalog;

    @Test
    void placingAnOrderStartsTheSaga() throws Exception {
        given(catalog.product(1L)).willReturn(new ProductView(1L, "Keyboard", 4900));
        mvc.perform(post("/api/orders").contentType(MediaType.APPLICATION_JSON)

                .andExpect(jsonPath("$.status").value("PENDING"));
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() ->
                assertThat(published.events).anyMatch(event -> event.startsWith("orders.placed")));

        Order order = orders.create(1, 2, 9800);
        kafka.send("stock.reserved", new StockReserved(order.id()));

                assertThat(orders.findById(order.id())).get().extracting(Order::status).isEqualTo(OrderStatus.CONFIRMED));

        kafka.send("stock.rejected", new StockRejected(order.id()));
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() ->
                assertThat(orders.findById(order.id())).get().extracting(Order::status).isEqualTo(OrderStatus.CANCELLED));
~~~
