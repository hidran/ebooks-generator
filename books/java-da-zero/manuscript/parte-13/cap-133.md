# Lezione 108. Tracing distribuito end to end

## Punti chiave

### Una richiesta, tre servizi

- Ordini chiama il catalogo via HTTP
- Ordini pubblica orders.placed, le notifiche lo consumano
- Tre processi, tre log, nessun filo comune
- Obiettivo: lo stesso trace id dall'ingresso all'ultimo listener

### Tracce, span e traceparent

- Traccia: il viaggio completo di una richiesta
- Span: una singola operazione, con padre e durata
- traceparent (W3C): trasporta il trace id tra i processi
- Micrometer Tracing + OpenTelemetry, esportazione OTLP
- HTTP automatico; Kafka: observation-enabled (template, listener)

> Percorso: Ordini, poi Catalogo (HTTP) / Kafka, poi Notifiche, poi Jaeger.

### Che cosa non serve (KISS)

- Nessuno span scritto a mano per i passaggi tra servizi
- In produzione si campiona: probability 1.0 solo in demo
- Jaeger in memoria solo in demo: in produzione un archivio
- Configurazione comune: nel Config Server, non ripetuta

`order-service/src/main/java/com/shop/orders/OrderService.java`

~~~java
package com.shop.orders;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;

@Service
public class OrderService {
    private static final Logger log = LoggerFactory.getLogger(OrderService.class);
    private final CatalogClient catalog;
    private final OrderRepository orders;
    private final KafkaTemplate<String, Object> events;

    public OrderService(CatalogClient catalog, OrderRepository orders, KafkaTemplate<String, Object> events) {
        this.catalog = catalog;
        this.orders = orders;
        this.events = events;
    }

    public Order place(long productId, int quantity) {
        ProductView product = catalog.product(productId);
        Order order = orders.save(productId, quantity, product.priceInCents() * quantity);
        events.send("orders.placed", String.valueOf(order.id()),
                new OrderPlaced(order.id(), productId, quantity, order.totalInCents()));
        log.info("Order {} placed", order.id());
        return order;
    }

    public Order find(long id) {
        return orders.findById(id).orElseThrow(() -> new OrderNotFoundException(id));
    }
}
~~~

~~~java
package com.shop.orders;

import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.io.UncheckedIOException;
import java.net.InetSocketAddress;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
import org.apache.kafka.clients.consumer.ConsumerRecord;
import org.apache.kafka.common.header.Header;
import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.micrometer.tracing.test.autoconfigure.AutoConfigureTracing;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.context.TestConfiguration;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.kafka.test.context.EmbeddedKafka;
import org.springframework.test.context.DynamicPropertyRegistry;
import org.springframework.test.context.DynamicPropertySource;
import org.springframework.test.web.servlet.MockMvc;
import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest(properties = {"spring.kafka.bootstrap-servers=${spring.embedded.kafka.brokers}",
        "management.tracing.export.otlp.enabled=false"})
@EmbeddedKafka(partitions = 1, topics = "orders.placed")
@AutoConfigureMockMvc
@AutoConfigureTracing
class TracePropagationTests {
    static final String TRACE_ID = "4bf92f3577b34da6a3ce929d0e0e4736";
    static final String TRACEPARENT = "00-" + TRACE_ID + "-00f067aa0ba902b7-01";
    static final List<String> CATALOG_TRACEPARENTS = new CopyOnWriteArrayList<>();
    static final HttpServer CATALOG = catalogStub();

    @Autowired
    MockMvc mvc;

    @Autowired
    PublishedEvents published;

    @DynamicPropertySource
    static void catalog(DynamicPropertyRegistry registry) {
        registry.add("spring.http.serviceclient.catalog.base-url",
                () -> "http://127.0.0.1:" + CATALOG.getAddress().getPort());
    }

    @AfterAll
    static void stopStub() {
        CATALOG.stop(0);
    }

    @Test
    void theCatalogCallCarriesTheSameTraceId() throws Exception {
        placeOrder();
        assertThat(CATALOG_TRACEPARENTS).isNotEmpty().allSatisfy(header -> assertThat(header).contains(TRACE_ID));
    }

    @Test
    void theOrderPlacedEventCarriesTheSameTraceId() throws Exception {
        placeOrder();
        await().atMost(Duration.ofSeconds(10)).untilAsserted(() ->
                assertThat(published.traceparents).anySatisfy(header -> assertThat(header).contains(TRACE_ID)));
    }

    private void placeOrder() throws Exception {
        mvc.perform(post("/api/orders").header("traceparent", TRACEPARENT)
                        .contentType(MediaType.APPLICATION_JSON).content("{\"productId\":1,\"quantity\":2}"))
                .andExpect(status().isCreated());
    }

    private static HttpServer catalogStub() {
        try {
            HttpServer server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
            server.createContext("/api/products/1", exchange -> {
                CATALOG_TRACEPARENTS.add(String.valueOf(exchange.getRequestHeaders().getFirst("traceparent")));
                byte[] body = "{\"id\":1,\"name\":\"Keyboard\",\"priceInCents\":4900}".getBytes(StandardCharsets.UTF_8);
                exchange.getResponseHeaders().add("Content-Type", "application/json");
                exchange.sendResponseHeaders(200, body.length);
                exchange.getResponseBody().write(body);
                exchange.close();
            });
            server.start();
            return server;
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }

    @TestConfiguration
    static class PublishedEvents {
        final List<String> traceparents = new CopyOnWriteArrayList<>();

        @KafkaListener(topics = "orders.placed", groupId = "orders-test")
        void capture(ConsumerRecord<String, String> record) {
            Header header = record.headers().lastHeader("traceparent");
            traceparents.add(header == null ? "none" : new String(header.value(), StandardCharsets.UTF_8));
        }
    }
}
~~~
