# Lezione 104. API Gateway con Spring Cloud Gateway

## Punti chiave

### Due indirizzi e codice ripetuto

- Il client conosce la porta del catalogo e quella degli ordini
- Ogni servizio copia lo stesso filtro X-Request-Id
- Una modifica va ripetuta in ogni servizio
- DRY violato da un aspetto trasversale

### Il gateway: rotte e filtri

- Un solo indirizzo per il client: 127.0.0.1:21043
- Rotta: predicato sul percorso e URI di destinazione
- Filtro: codice che avvolge ogni richiesta inoltrata
- Spring Cloud Gateway Server Web MVC (train 2025.1.3)

> Percorso: Client, poi Gateway, poi Catalogo / Ordini.

### Quando il gateway non serve (KISS)

- Un solo servizio: nessun vantaggio, un processo in più
- Kubernetes: un ingress può già instradare
- Nessuna logica di dominio nel gateway
- Solo aspetti comuni: identificativi, sicurezza, limiti

`gateway/src/main/java/com/shop/gateway/RequestIdFilter.java`

~~~java
package com.shop.gateway;

import java.util.UUID;
import org.springframework.web.servlet.function.HandlerFilterFunction;
import org.springframework.web.servlet.function.ServerRequest;
import org.springframework.web.servlet.function.ServerResponse;

public final class RequestIdFilter {
    public static final String HEADER = "X-Request-Id";

    private RequestIdFilter() {
    }
    private static String idFrom(ServerRequest request) {
        String incoming = request.headers().firstHeader(HEADER);
        return incoming == null || incoming.isBlank() ? UUID.randomUUID().toString() : incoming;
    }
    private static ServerRequest withId(ServerRequest request, String requestId) {
        return ServerRequest.from(request)
                .headers(headers -> headers.set(HEADER, requestId)).build();
    }
    public static HandlerFilterFunction<ServerResponse, ServerResponse> requestId() {
        return (request, next) -> {
            String requestId = idFrom(request);
            ServerResponse response = next.handle(withId(request, requestId));
            response.headers().set(HEADER, requestId);
            return response;
        };
    }
}
~~~

`gateway/src/main/java/com/shop/gateway/GatewayRoutes.java`

~~~java
package com.shop.gateway;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.servlet.function.RouterFunction;
import org.springframework.web.servlet.function.ServerResponse;
import static org.springframework.cloud.gateway.server.mvc.filter.BeforeFilterFunctions.uri;
import static org.springframework.cloud.gateway.server.mvc.handler.GatewayRouterFunctions.route;
import static org.springframework.cloud.gateway.server.mvc.handler.HandlerFunctions.http;
import static org.springframework.web.servlet.function.RequestPredicates.path;

@Configuration
public class GatewayRoutes {
    @Bean
    RouterFunction<ServerResponse> catalogRoute(@Value("${shop.catalog-url}") String catalogUrl) {
        return route("catalog").route(path("/api/products/**"), http()).before(uri(catalogUrl))
                .filter(RequestIdFilter.requestId()).build();
    }
    @Bean
    RouterFunction<ServerResponse> ordersRoute(@Value("${shop.orders-url}") String ordersUrl) {
        return route("orders").route(path("/api/orders/**"), http()).before(uri(ordersUrl))
                .filter(RequestIdFilter.requestId()).build();
    }
}
~~~

~~~java
public class ProductNotFoundException extends ErrorResponseException {
    public ProductNotFoundException(long id) {
        super(HttpStatus.NOT_FOUND, problem(id), null);

        problem.setType(URI.create("https://api.shop.example/problems/product-not-found"));
        problem.setTitle("Product not found");
        return problem;
    }
}
~~~

~~~java
package com.shop.gateway;

import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.io.UncheckedIOException;
import java.net.InetSocketAddress;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.Test;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.web.server.LocalServerPort;
import org.springframework.test.context.DynamicPropertyRegistry;
import org.springframework.test.context.DynamicPropertySource;
import static org.assertj.core.api.Assertions.assertThat;

@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
class GatewayRoutesTests {
    static final HttpServer CATALOG = stub("catalog");
    static final HttpServer ORDERS = stub("orders");

    @LocalServerPort
    int port;

    @DynamicPropertySource
    static void services(DynamicPropertyRegistry registry) {
        registry.add("shop.catalog-url", () -> "http://127.0.0.1:" + CATALOG.getAddress().getPort());
        registry.add("shop.orders-url", () -> "http://127.0.0.1:" + ORDERS.getAddress().getPort());
    }

    @AfterAll
    static void stopStubs() {
        CATALOG.stop(0);
        ORDERS.stop(0);
    }

    @Test
    void routesProductsToTheCatalog() throws Exception {
        HttpResponse<String> response = get("/api/products/1", null);
        assertThat(response.statusCode()).isEqualTo(200);
        assertThat(response.body()).startsWith("catalog GET /api/products/1");
    }

    @Test
    void routesOrdersToTheOrderService() throws Exception {
        HttpResponse<String> response = get("/api/orders/7", null);
        assertThat(response.statusCode()).isEqualTo(200);
        assertThat(response.body()).startsWith("orders GET /api/orders/7");
    }

    @Test
    void addsARequestIdWhenTheClientSendsNone() throws Exception {
        HttpResponse<String> response = get("/api/products/1", null);
        assertThat(response.statusCode()).isEqualTo(200);
        String requestId = response.headers().firstValue("X-Request-Id").orElse("");
        assertThat(requestId).hasSize(36);
        assertThat(response.body()).endsWith("request-id=" + requestId);
    }

    @Test
    void keepsTheRequestIdSentByTheClient() throws Exception {
        HttpResponse<String> response = get("/api/orders/7", "demo-42");
        assertThat(response.statusCode()).isEqualTo(200);
        assertThat(response.headers().firstValue("X-Request-Id")).hasValue("demo-42");
        assertThat(response.body()).endsWith("request-id=demo-42");
    }

    private HttpResponse<String> get(String path, String requestId) throws Exception {
        HttpRequest.Builder request = HttpRequest.newBuilder(URI.create("http://127.0.0.1:" + port + path));
        if (requestId != null) {
            request.header("X-Request-Id", requestId);
        }
        return HttpClient.newHttpClient().send(request.build(), HttpResponse.BodyHandlers.ofString());
    }

    private static HttpServer stub(String service) {
        try {
            HttpServer server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
            server.createContext("/", exchange -> {
                String text = service + " " + exchange.getRequestMethod() + " " + exchange.getRequestURI()
                        + " request-id=" + exchange.getRequestHeaders().getFirst("X-Request-Id");
                byte[] body = text.getBytes(StandardCharsets.UTF_8);
                exchange.sendResponseHeaders(200, body.length);
                exchange.getResponseBody().write(body);
                exchange.close();
            });
            server.start();
            return server;
        } catch (IOException e) {
            throw new UncheckedIOException(e);
        }
    }
}
~~~

~~~java
package com.shop.orders;

import java.net.URI;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/orders")
public class OrderController {
    private final OrderService service;

    public OrderController(OrderService service) {
        this.service = service;
    }

    @PostMapping
    public ResponseEntity<OrderResponse> place(@RequestBody PlaceOrderRequest request) {
        Order order = service.place(request.productId(), request.quantity());
        return ResponseEntity.created(URI.create("/api/orders/" + order.id())).body(OrderResponse.from(order));
    }

    @GetMapping("/{id}")
    public OrderResponse order(@PathVariable long id) {
        return OrderResponse.from(service.find(id));
    }
}
~~~

~~~java
package com.shop.orders;

import java.util.Map;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;
import org.springframework.stereotype.Repository;

@Repository
public class OrderRepository {
    private final Map<Long, Order> orders = new ConcurrentHashMap<>();
    private final AtomicLong nextId = new AtomicLong(1);

    public Order save(long productId, int quantity, int totalInCents) {
        Order order = new Order(nextId.getAndIncrement(), productId, quantity, totalInCents);
        orders.put(order.id(), order);
        return order;
    }

    public Optional<Order> findById(long id) {
        return Optional.ofNullable(orders.get(id));
    }
}
~~~

~~~java
package com.shop.orders;

import org.springframework.stereotype.Service;

@Service
public class OrderService {
    private final CatalogClient catalog;
    private final OrderRepository orders;

    public OrderService(CatalogClient catalog, OrderRepository orders) {
        this.catalog = catalog;
        this.orders = orders;
    }

    public Order place(long productId, int quantity) {
        ProductView product = catalog.product(productId);
        return orders.save(productId, quantity, product.priceInCents() * quantity);
    }

    public Order find(long id) {
        return orders.findById(id).orElseThrow(() -> new OrderNotFoundException(id));
    }
}
~~~
