# Lezione 69. Da JDBC a JPA

## Punti chiave

### Dalla memoria al database

- H2 in memoria: nessuna installazione
- schema.sql crea la tabella, data.sql i dati di prova
- JdbcClient: SQL scritto da noi, parametri con nome
- Mapping manuale: da ResultSet a record

### Cosa risolve un ORM

- Mapping tra classe e tabella dichiarato una volta
- INSERT, UPDATE e SELECT generati
- Modifiche agli oggetti scritte in automatico
- Relazioni e differenze tra database
- JPA è lo standard, Hibernate l'implementazione

### Cosa costa un ORM

- L'SQL generato va controllato
- Una relazione può generare molte query
- Ciclo di vita delle entità e proxy da capire
- Curva di apprendimento reale

### KISS: quando JDBC basta

- Report e query complesse
- Importazioni massive
- Poche tabelle e logica semplice
- La porta ProductRepository nasconde la scelta

`src/main/java/com/shop/catalog/JdbcProductRepository.java`

~~~java
package com.shop.catalog;

import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.List;
import java.util.Optional;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.jdbc.support.GeneratedKeyHolder;
import org.springframework.jdbc.support.KeyHolder;
import org.springframework.stereotype.Repository;

@Repository
public class JdbcProductRepository implements ProductRepository {
    private final JdbcClient jdbc;

    public JdbcProductRepository(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    @Override
    public List<Product> findAll() {
        return jdbc.sql("SELECT id, name, price_in_cents FROM product ORDER BY id")
                .query(JdbcProductRepository::toProduct)
                .list();
    }

    @Override
    public Optional<Product> findById(long id) {
        return jdbc.sql("SELECT id, name, price_in_cents FROM product WHERE id = :id")
                .param("id", id)
                .query(JdbcProductRepository::toProduct)
                .optional();
    }

    @Override
    public Product save(Product product) {
        KeyHolder keys = new GeneratedKeyHolder();
        jdbc.sql("INSERT INTO product (name, price_in_cents) VALUES (:name, :price)")
                .param("name", product.name())
                .param("price", product.priceInCents())
                .update(keys, "id");
        long id = keys.getKeyAs(Long.class);
        return new Product(id, product.name(), product.priceInCents());
    }

    @Override
    public boolean deleteById(long id) {
        return jdbc.sql("DELETE FROM product WHERE id = :id").param("id", id).update() == 1;
    }

    private static Product toProduct(ResultSet row, int rowNumber) throws SQLException {
        return new Product(row.getLong("id"), row.getString("name"), row.getInt("price_in_cents"));
    }
}
~~~

`src/test/java/com/shop/catalog/JdbcProductRepositoryTests.java`

~~~java
package com.shop.catalog;

import static org.assertj.core.api.Assertions.assertThat;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.jdbc.test.autoconfigure.JdbcTest;
import org.springframework.context.annotation.Import;

@JdbcTest
@Import(JdbcProductRepository.class)
class JdbcProductRepositoryTests {
    @Autowired
    JdbcProductRepository repository;

    @Test
    void findAllMapsEveryRow() {
        assertThat(repository.findAll()).containsExactly(
                new Product(1, "Keyboard", 4900),
                new Product(2, "Mouse", 2500),
                new Product(3, "Monitor", 18900));
    }

    @Test
    void findByIdMapsOneRow() {
        assertThat(repository.findById(2)).contains(new Product(2, "Mouse", 2500));
    }

    @Test
    void findByIdOfAMissingRowIsEmpty() {
        assertThat(repository.findById(999)).isEmpty();
    }

    @Test
    void saveReturnsTheGeneratedId() {
        Product saved = repository.save(new Product(0, "Webcam", 3900));
        assertThat(saved.id()).isPositive();
        assertThat(repository.findAll()).contains(saved);
    }

    @Test
    void deleteByIdRemovesTheRow() {
        assertThat(repository.deleteById(3)).isTrue();
        assertThat(repository.deleteById(3)).isFalse();
    }
}
~~~
