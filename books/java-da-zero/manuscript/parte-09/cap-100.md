# Lezione 77. Auditing e optimistic locking

## Punti chiave

### Due problemi

- Quando è stato creato o modificato un prodotto?
- Due client leggono la stessa versione
- Il secondo salvataggio cancella il primo
- Aggiornamento perso: nessun errore, dati sbagliati

> Percorso: Alice legge v0, poi Bob legge v0, poi Alice salva v1, poi Bob sovrascrive.

### Come funziona @Version

- UPDATE ... WHERE id = ? AND version = ?
- Versione cambiata: ObjectOptimisticLockingFailureException
- Nessun lock mentre l'utente decide
- Il client rilegge e riprova: 409 Conflict

### KISS: cosa non serve

- Niente lock pessimistici per richieste web brevi
- Niente tabelle di audit scritte a mano per due date
- Niente timestamp impostati a mano nel servizio
- Chi ha modificato: @CreatedBy e AuditorAware, quando servirà

~~~java
package com.shop.catalog;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EntityListeners;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import jakarta.persistence.Version;
import java.time.Instant;
import org.springframework.data.annotation.CreatedDate;
import org.springframework.data.annotation.LastModifiedDate;
import org.springframework.data.jpa.domain.support.AuditingEntityListener;

@Entity
@Table(name = "product")
@EntityListeners(AuditingEntityListener.class)
public class ProductEntity {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @Column(nullable = false, length = 100)
    private String name;
    private int priceInCents;
    @Version
    private long version;
    @CreatedDate
    @Column(updatable = false)
    private Instant createdAt;
    @LastModifiedDate
    private Instant updatedAt;

    protected ProductEntity() {
    }

    private ProductEntity(Long id, String name, int priceInCents, long version) {
        this.id = id;
        this.name = name;
        this.priceInCents = priceInCents;
        this.version = version;
    }

    static ProductEntity from(Product product) {
        Long id = product.id() == 0 ? null : product.id();
        return new ProductEntity(id, product.name(), product.priceInCents(), product.version());
    }

    Product toProduct() {
        return new Product(id, name, priceInCents, version);
    }

    Instant getCreatedAt() {
        return createdAt;
    }

    Instant getUpdatedAt() {
        return updatedAt;
    }
}
~~~

~~~java
package com.shop.web;

import com.shop.catalog.ProductNotFoundException;
import java.net.URI;
import java.util.Map;
import java.util.TreeMap;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.orm.ObjectOptimisticLockingFailureException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

@RestControllerAdvice
public class ApiErrorHandler extends ResponseEntityExceptionHandler {
    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(MethodArgumentNotValidException exception,
            HttpHeaders headers, HttpStatusCode status, WebRequest request) {
        Map<String, String> errors = new TreeMap<>();
        exception.getBindingResult().getFieldErrors()
                .forEach(error -> errors.putIfAbsent(error.getField(), error.getDefaultMessage()));
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(status, "One or more fields are invalid.");
        problem.setType(URI.create("https://api.shop.example/problems/validation-error"));
        problem.setTitle("Request validation failed");
        problem.setProperty("errors", errors);
        return handleExceptionInternal(exception, problem, headers, status, request);
    }

    @ExceptionHandler(ProductNotFoundException.class)
    public ProblemDetail handleNotFound(ProductNotFoundException exception) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, exception.getMessage());
        problem.setType(URI.create("https://api.shop.example/problems/product-not-found"));
        problem.setTitle("Product not found");
        return problem;
    }
    @ExceptionHandler(ObjectOptimisticLockingFailureException.class)
    public ProblemDetail handleConflict() {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.CONFLICT,
                "The product was changed by someone else. Read it again and retry.");
        problem.setType(URI.create("https://api.shop.example/problems/concurrent-update"));
        problem.setTitle("Concurrent update");
        return problem;
    }
}
~~~

`src/main/java/com/shop/catalog/JpaProductRepository.java`

~~~java
package com.shop.catalog;

import java.util.List;
import java.util.Optional;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Repository;

@Repository
public class JpaProductRepository implements ProductRepository {
    private final ProductJpaRepository jpa;
    public JpaProductRepository(ProductJpaRepository jpa) {
        this.jpa = jpa;
    }
    @Override
    public List<Product> findAll() {
        return jpa.findAll(Sort.by("id")).stream().map(ProductEntity::toProduct).toList();
    }
    @Override
    public Optional<Product> findById(long id) {
        return jpa.findById(id).map(ProductEntity::toProduct);
    }
    @Override
    public Product save(Product product) {
        return jpa.saveAndFlush(ProductEntity.from(product)).toProduct();
    }
    @Override
    public boolean deleteById(long id) {
        if (!jpa.existsById(id)) {
            return false;
        }
        jpa.deleteById(id);
        return true;
    }
}
~~~

~~~java
package com.shop.catalog;

import java.util.List;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private final ProductRepository repository;

    public ProductService(ProductRepository repository) {
        this.repository = repository;
    }

    public List<Product> list() {
        return repository.findAll();
    }

    public Product find(long id) {
        return repository.findById(id).orElseThrow(() -> new ProductNotFoundException(id));
    }

    public Product create(String name, int priceInCents) {
        return repository.save(new Product(0, name, priceInCents, 0));
    }

    public Product update(long id, String name, int priceInCents, long version) {
        find(id);
        return repository.save(new Product(id, name, priceInCents, version));
    }

    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

`src/main/java/com/shop/web/ProductController.java`

~~~java
package com.shop.web;

import com.shop.catalog.Product;
import com.shop.catalog.ProductService;
import jakarta.validation.Valid;
import java.net.URI;
import java.util.List;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/products")
public class ProductController {
    private final ProductService service;

    public ProductController(ProductService service) {
        this.service = service;
    }

    @GetMapping
    public List<ProductResponse> list() {
        return service.list().stream().map(ProductResponse::from).toList();
    }

    @GetMapping("/{id}")
    public ProductResponse find(@PathVariable long id) {
        return ProductResponse.from(service.find(id));
    }

    @PostMapping
    public ResponseEntity<ProductResponse> create(@Valid @RequestBody CreateProductRequest request) {
        Product product = service.create(request.name(), request.priceInCents());
        URI location = URI.create("/api/products/" + product.id());
        return ResponseEntity.created(location).body(ProductResponse.from(product));
    }

    @PutMapping("/{id}")
    public ProductResponse update(@PathVariable long id, @Valid @RequestBody UpdateProductRequest request) {
        return ProductResponse.from(
                service.update(id, request.name(), request.priceInCents(), request.version()));
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void delete(@PathVariable long id) {
        service.delete(id);
    }
}
~~~

~~~java
package com.shop.catalog;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.orm.ObjectOptimisticLockingFailureException;

@SpringBootTest
class OptimisticLockingTests {
    @Autowired
    ProductService service;

    @Test
    void updateWithTheCurrentVersionIsAccepted() {
        Product created = service.create("Headset", 5900);
        Product updated = service.update(created.id(), "Headset", 5400, created.version());
        assertThat(updated.priceInCents()).isEqualTo(5400);
    }

    @Test
    void updateWithAStaleVersionIsRejected() {
        Product created = service.create("Speaker", 7900);
        service.update(created.id(), "Speaker", 7500, created.version());
        assertThatThrownBy(() -> service.update(created.id(), "Speaker", 6900, created.version()))
                .isInstanceOf(ObjectOptimisticLockingFailureException.class);
    }
}
~~~

`src/test/java/com/shop/catalog/ProductAuditingTests.java`

~~~java
package com.shop.catalog;

import static org.assertj.core.api.Assertions.assertThat;

import java.time.Instant;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

@SpringBootTest
class ProductAuditingTests {
    @Autowired
    ProductService service;
    @Autowired
    ProductJpaRepository entities;

    @Test
    void createSetsBothTimestamps() {
        Product created = service.create("Webcam", 3900);
        ProductEntity entity = entities.findById(created.id()).orElseThrow();
        assertThat(entity.getCreatedAt()).isNotNull();
        assertThat(entity.getUpdatedAt()).isNotNull();
    }

    @Test
    void updateKeepsCreatedAtAndMovesUpdatedAt() {
        Product created = service.create("Lamp", 2900);
        Instant createdAt = entities.findById(created.id()).orElseThrow().getCreatedAt();
        service.update(created.id(), "Desk lamp", 3100, created.version());
        ProductEntity updated = entities.findById(created.id()).orElseThrow();
        assertThat(createdAt).isNotNull();
        assertThat(updated.getCreatedAt()).isEqualTo(createdAt);
        assertThat(updated.getUpdatedAt()).isAfterOrEqualTo(createdAt);
    }
}
~~~
