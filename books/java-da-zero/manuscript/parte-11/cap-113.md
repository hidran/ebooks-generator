# Lezione 90. Resilienza: timeout, retry, circuit breaker

## Punti chiave

### Timeout

- Senza timeout un fornitore lento blocca i nostri thread
- spring.http.clients.connect-timeout: 500ms
- spring.http.clients.read-timeout: 1s
- Valgono per i RestClient creati dal builder di Boot
- Timeout: ResourceAccessException, risposta 503 ProblemDetail

### Retry in Spring Framework 7

- @Retryable(includes, maxRetries = 2, delay = 200): tre tentativi
- Solo errori temporanei del server, non timeout né 4xx
- Solo operazioni idempotenti, come una lettura
- @EnableResilientMethods è obbligatoria

### Circuit breaker e limiti

- Chiuso, aperto, semiaperto: dopo troppi errori risponde subito
- Resilience4j: resilience4j-spring-boot3 2.4.0 rifiuta Spring Boot 4
- @ConcurrencyLimit(2): al massimo due chiamate insieme
- Le chiamate in più aspettano (BLOCK) o vengono rifiutate (REJECT)

### Prima e dopo: KISS e DRY

- Prima: ciclo, try catch, ultimo errore, pausa: 21 righe
- Dopo: due annotazioni e una chiamata: 6 righe
- DRY: la politica non si copia in ogni client
- KISS: nessuna libreria in più finché basta il framework

~~~java
package com.shop.supplier;

import com.shop.stock.StockLevel;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class FakeSupplierController {
    private static final Logger LOGGER = LoggerFactory.getLogger(FakeSupplierController.class);
    private final Map<Long, AtomicInteger> calls = new ConcurrentHashMap<>();

    @GetMapping("/supplier/stock/{productId}")
    public ResponseEntity<StockLevel> stock(@PathVariable long productId)
            throws InterruptedException {
        int call = calls.computeIfAbsent(productId, id -> new AtomicInteger()).incrementAndGet();
        if (productId == 2) {
            LOGGER.info("Supplier is slow for product {}", productId);
            Thread.sleep(3000);
        }
        if (productId == 4 || (productId == 3 && call % 3 != 0)) {
            LOGGER.info("Supplier fails for product {}, call {}", productId, call);
            return ResponseEntity.status(HttpStatus.SERVICE_UNAVAILABLE).build();
        }
        LOGGER.info("Supplier answers for product {}, call {}", productId, call);
        return ResponseEntity.ok(new StockLevel(productId, 12));
    }

    public int calls(long productId) {
        return calls.getOrDefault(productId, new AtomicInteger()).get();
    }
}
~~~

`src/main/java/com/shop/stock/SupplierClient.java`

~~~java
package com.shop.stock;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.resilience.annotation.ConcurrencyLimit;
import org.springframework.resilience.annotation.Retryable;
import org.springframework.stereotype.Component;
import org.springframework.web.client.HttpServerErrorException;
import org.springframework.web.client.RestClient;

@Component
public class SupplierClient {
    private final RestClient restClient;

    public SupplierClient(RestClient.Builder builder,
            @Value("${shop.supplier.base-url}") String baseUrl) {
        this.restClient = builder.baseUrl(baseUrl).build();
    }

    @Retryable(includes = HttpServerErrorException.class, maxRetries = 2, delay = 200)
    @ConcurrencyLimit(2)
    public StockLevel stock(long productId) {
        return restClient.get().uri("/stock/{id}", productId)
                .retrieve().body(StockLevel.class);
    }
}
~~~

~~~java
package com.shop.web;

import java.net.URI;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.client.RestClientException;

@RestControllerAdvice
public class ApiErrorHandler {
    @ExceptionHandler(RestClientException.class)
    public ProblemDetail supplierUnavailable(RestClientException exception) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(
                HttpStatus.SERVICE_UNAVAILABLE, "The supplier is not available right now.");
        problem.setType(URI.create("https://api.shop.example/problems/supplier-unavailable"));
        problem.setTitle("Supplier unavailable");
        return problem;
    }
}
~~~

`src/test/java/com/shop/SupplierClientTests.java`

~~~java
package com.shop;

import com.shop.stock.SupplierClient;
import com.shop.supplier.FakeSupplierController;
import java.time.Duration;
import java.time.Instant;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.context.SpringBootTest.WebEnvironment;
import org.springframework.web.client.HttpServerErrorException;
import org.springframework.web.client.ResourceAccessException;
import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

@SpringBootTest(webEnvironment = WebEnvironment.DEFINED_PORT)
class SupplierClientTests {
    @Autowired SupplierClient client;
    @Autowired FakeSupplierController supplier;

    @Test
    void answersWithTheStock() {
        assertThat(client.stock(1).available()).isEqualTo(12);
    }

    @Test
    void flakySupplierIsRetried() {
        assertThat(client.stock(3).available()).isEqualTo(12);
        assertThat(supplier.calls(3)).isEqualTo(3);
    }

    @Test
    void failingSupplierGivesUpAfterThreeAttempts() {
        assertThatThrownBy(() -> client.stock(4)).isInstanceOf(HttpServerErrorException.class);
        assertThat(supplier.calls(4)).isEqualTo(3);
    }

    @Test
    void slowSupplierTimesOut() {
        Instant start = Instant.now();
        assertThatThrownBy(() -> client.stock(2)).isInstanceOf(ResourceAccessException.class);
        assertThat(Duration.between(start, Instant.now())).isLessThan(Duration.ofSeconds(2));
    }
}
~~~
