# Lezione 88. Tracing e log correlati

## Punti chiave

### Trace e span

- Trace id: 32 cifre esadecimali, uno per richiesta
- Span id: 16 cifre esadecimali, uno per operazione
- Header W3C traceparent: la traccia continua fra servizi
- Gli identificativi stanno nel contesto dei log (MDC)

### Micrometer Tracing in Spring Boot 4

- spring-boot-starter-opentelemetry
- Micrometer Tracing con il bridge OpenTelemetry
- Log di Boot: [traceId-spanId] su ogni riga della richiesta
- Nessun collector necessario per vedere gli identificativi

### Campionamento ed esportazione

- management.tracing.sampling.probability: 0.1 per default
- 1.0 nella demo: ogni traccia registrata
- Gli span si esportano solo con un endpoint OTLP configurato
- Metriche OTLP disattivate: nessun collector in ascolto

### Prima e dopo: DRY

- Prima: un requestId passato a mano a ogni metodo
- Prima: lo stesso id concatenato in ogni messaggio
- Dopo: il contesto della traccia vive nell'MDC
- Dopo: il pattern dei log aggiunge gli id una volta sola
- KISS: niente filtri scritti a mano, niente collector per iniziare

`src/main/java/com/shop/catalog/ProductService.java`

~~~java
package com.shop.catalog;

import java.util.List;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private static final Logger LOGGER = LoggerFactory.getLogger(ProductService.class);
    private final ProductRepository repository;

    public ProductService(ProductRepository repository) {
        this.repository = repository;
    }

    public List<Product> list() {
        return repository.findAll();
    }

    public Product find(long id) {
        LOGGER.info("Finding product {}", id);
        Product product = repository.findById(id)
                .orElseThrow(() -> new ProductNotFoundException(id));
        LOGGER.info("Found product {}", product.name());
        return product;
    }

    public Product create(String name, int priceInCents) {
        return repository.save(new Product(0, name, priceInCents));
    }

    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

`src/test/java/com/shop/TracingTests.java`

~~~java
package com.shop;

import java.util.regex.Matcher;
import java.util.regex.Pattern;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.system.CapturedOutput;
import org.springframework.boot.test.system.OutputCaptureExtension;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.test.web.servlet.MockMvc;
import static org.assertj.core.api.Assertions.assertThat;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
@ExtendWith(OutputCaptureExtension.class)
class TracingTests {
    private static final String TRACE_ID = "4bf92f3577b34da6a3ce929d0e0e4736";
    private static final String SPAN_AND_TEXT = "-[0-9a-f]{16}\\] .*";

    @Autowired MockMvc mvc;

    @Test
    void productIsReturned() throws Exception {
        mvc.perform(get("/api/products/1"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.name").value("Keyboard"));
    }

    @Test
    void incomingTraceparentIsContinued(CapturedOutput output) throws Exception {
        String traceparent = "00-" + TRACE_ID + "-00f067aa0ba902b7-01";
        mvc.perform(get("/api/products/2").header("traceparent", traceparent))
                .andExpect(status().isOk());
        String found = TRACE_ID + SPAN_AND_TEXT + "Found product Mouse";
        assertThat(output.getOut()).containsPattern(found);
    }

    @Test
    void bothLogLinesShareOneTraceId(CapturedOutput output) throws Exception {
        mvc.perform(get("/api/products/3")).andExpect(status().isOk());
        String start = "\\[([0-9a-f]{32})" + SPAN_AND_TEXT + "Finding product 3";
        Matcher finding = Pattern.compile(start).matcher(output.getOut());
        assertThat(finding.find()).isTrue();
        String found = finding.group(1) + SPAN_AND_TEXT + "Found product Monitor";
        assertThat(output.getOut()).containsPattern(found);
    }
}
~~~
