# Lezione 89. Caching con Spring Cache e Caffeine

## Punti chiave

### Spring Cache

- Enable Caching attiva un proxy intorno ai bean
- Cacheable: prima chiamata eseguita, poi risultato dalla cache
- Chiave: il parametro id ("#id")
- Chiamate interne alla stessa classe saltano il proxy

### Caffeine

- Cache in memoria, una per istanza
- spring.cache.cache-names: products, creata all'avvio
- maximumSize=500, expireAfterWrite=10m
- recordStats: metriche cache.gets con result hit e miss

### Invalidazione corretta

- find: Cacheable
- update: CachePut scrive il valore restituito
- delete: CacheEvict toglie la voce
- Stessa cache e stessa chiave su tutti e tre

### I rischi (KISS)

- Dati vecchi: ogni istanza ha la sua cache
- Modifiche fuori dal servizio non invalidano la cache
- Sempre una scadenza come rete di sicurezza
- In cache solo letture frequenti e misurate
- Cache distribuita solo quando serve davvero

~~~java
package com.shop;

import com.shop.catalog.Product;
import com.shop.catalog.ProductNotFoundException;
import com.shop.catalog.ProductRepository;
import com.shop.catalog.ProductService;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.cache.CacheManager;
import org.springframework.cache.caffeine.CaffeineCache;
import org.springframework.test.context.bean.override.mockito.MockitoSpyBean;
import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;
import static org.mockito.Mockito.times;
import static org.mockito.Mockito.verify;

@SpringBootTest
class ProductCacheTests {
    @Autowired ProductService service;
    @Autowired CacheManager cacheManager;
    @MockitoSpyBean ProductRepository repository;

    @BeforeEach
    void emptyTheCache() {
        cacheManager.getCache("products").clear();
    }

    @Test
    void secondReadComesFromTheCache() {
        service.find(1);
        service.find(1);
        verify(repository, times(1)).findById(1L);
    }

    @Test
    void statisticsCountTheHit() {
        CaffeineCache cache = (CaffeineCache) cacheManager.getCache("products");
        long hits = cache.getNativeCache().stats().hitCount();
        service.find(2);
        service.find(2);
        assertThat(cache.getNativeCache().stats().hitCount()).isEqualTo(hits + 1);
    }

    @Test
    void updateReplacesTheCachedProduct() {
        service.find(3);
        service.update(3, "Curved monitor", 21900);
        assertThat(service.find(3).name()).isEqualTo("Curved monitor");
    }

    @Test
    void deleteEvictsTheCachedProduct() {
        Product webcam = service.create("Webcam", 3900);
        service.find(webcam.id());
        service.delete(webcam.id());
        assertThatThrownBy(() -> service.find(webcam.id()))
                .isInstanceOf(ProductNotFoundException.class);
    }
}
~~~

`src/main/java/com/shop/catalog/ProductService.java`

~~~java
package com.shop.catalog;

import java.util.List;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.cache.annotation.CacheEvict;
import org.springframework.cache.annotation.CachePut;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private static final Logger LOGGER = LoggerFactory.getLogger(ProductService.class);
    private final ProductRepository repository;

    public ProductService(ProductRepository repository) {
        this.repository = repository;
    }

    public List<Product> list() {
        return repository.findAll();
    }

    @Cacheable(cacheNames = "products", key = "#id")
    public Product find(long id) {
        LOGGER.info("Loading product {} from the repository", id);
        return repository.findById(id).orElseThrow(() -> new ProductNotFoundException(id));
    }

    public Product create(String name, int priceInCents) {
        return repository.save(new Product(0, name, priceInCents));
    }

    @CachePut(cacheNames = "products", key = "#id")
    public Product update(long id, String name, int priceInCents) {
        if (repository.findById(id).isEmpty()) {
            throw new ProductNotFoundException(id);
        }
        return repository.save(new Product(id, name, priceInCents));
    }

    @CacheEvict(cacheNames = "products", key = "#id")
    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

`src/main/java/com/shop/web/ProductController.java`

~~~java
package com.shop.web;

import com.shop.catalog.Product;
import com.shop.catalog.ProductService;
import jakarta.validation.Valid;
import java.net.URI;
import java.util.List;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/products")
public class ProductController {
    private final ProductService service;

    public ProductController(ProductService service) {
        this.service = service;
    }

    @GetMapping
    public List<ProductResponse> list() {
        return service.list().stream().map(ProductResponse::from).toList();
    }

    @GetMapping("/{id}")
    public ProductResponse find(@PathVariable long id) {
        return ProductResponse.from(service.find(id));
    }

    @PostMapping
    public ResponseEntity<ProductResponse> create(@Valid @RequestBody CreateProductRequest request) {
        Product product = service.create(request.name(), request.priceInCents());
        URI location = URI.create("/api/products/" + product.id());
        return ResponseEntity.created(location).body(ProductResponse.from(product));
    }

    @PutMapping("/{id}")
    public ProductResponse update(@PathVariable long id, @Valid @RequestBody UpdateProductRequest request) {
        return ProductResponse.from(service.update(id, request.name(), request.priceInCents()));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable long id) {
        service.delete(id);
        return ResponseEntity.noContent().build();
    }
}
~~~
