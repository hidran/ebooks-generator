# Lezione 92. Qualità del codice nel build

## Punti chiave

### Tre controlli, un comando

- Spotless: formato, fase validate
- Checkstyle: regole di stile, fase validate
- JaCoCo: copertura dei test, fase verify
- ./mvnw verify li esegue tutti, in ordine

### Spotless e Checkstyle

- spotless:check fallisce, spotless:apply corregge
- Passi testuali: spazi finali, a capo finale, niente tabulazioni
- checkstyle.xml: righe max 120, niente import con asterisco o inutilizzati
- Graffe obbligatorie, niente catch vuoti, costanti in maiuscolo
- Checkstyle indica la riga, la correzione è nostra

### JaCoCo

- prepare-agent registra le righe eseguite dai test
- report: target/site/jacoco/index.html
- check: righe coperte almeno 0.85
- Copertura alta non significa test buoni

### Che cosa non serve (KISS)

- Centinaia di regole copiate da altri progetti
- Una soglia al 100%
- Un formatter completo non provato con il proprio JDK
- Poche regole con un motivo condiviso dal team

`src/test/java/com/shop/ProductApiTests.java`

~~~java
package com.shop;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.delete;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class ProductApiTests {
    @Autowired MockMvc mvc;

    @Test
    void readsAProduct() throws Exception {
        mvc.perform(get("/api/products/1"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.name").value("Keyboard"));
    }

    @Test
    void createsAProduct() throws Exception {
        mvc.perform(post("/api/products").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\"Webcam\",\"priceInCents\":3900}"))
                .andExpect(status().isCreated());
    }

    @Test
    void rejectsAnInvalidProduct() throws Exception {
        mvc.perform(post("/api/products").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\" \",\"priceInCents\":0}"))
                .andExpect(status().isBadRequest())
                .andExpect(jsonPath("$.errors.name").value("must not be blank"));
    }

    @Test
    void deletedProductIsGone() throws Exception {
        mvc.perform(delete("/api/products/2")).andExpect(status().isNoContent());
        mvc.perform(get("/api/products/2"))
                .andExpect(status().isNotFound())
                .andExpect(jsonPath("$.title").value("Product not found"));
    }
}
~~~

~~~java
package com.shop;

import com.shop.catalog.InMemoryProductRepository;
import com.shop.catalog.Product;
import com.shop.catalog.ProductNotFoundException;
import com.shop.catalog.ProductService;
import org.junit.jupiter.api.Test;
import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class ProductServiceTests {
    private final ProductService service = new ProductService(new InMemoryProductRepository());

    @Test
    void listReturnsTheDemoProducts() {
        assertThat(service.list()).extracting(Product::name).containsExactly("Keyboard", "Mouse", "Monitor");
    }

    @Test
    void createAssignsTheNextId() {
        assertThat(service.create("Webcam", 3900).id()).isEqualTo(4);
    }

    @Test
    void findingAnUnknownProductFails() {
        assertThatThrownBy(() -> service.find(99)).isInstanceOf(ProductNotFoundException.class);
    }
}
~~~
