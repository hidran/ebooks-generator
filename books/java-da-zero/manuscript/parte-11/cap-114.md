# Lezione 91. Job schedulati e task asincroni

## Punti chiave

### Lavori pianificati

- @Scheduled(fixedDelay): pausa fissa dopo la fine di ogni esecuzione
- fixedRate: intervallo fisso dall'inizio, anche se l'esecuzione è lenta
- cron di Spring: sei campi, secondi minuti ore giorno mese giorno della settimana
- "0 0 2 * * *": ogni notte alle due
- @EnableScheduling avvia lo scheduler

### Task asincroni

- @Async: il metodo gira su un altro thread
- Restituisce CompletableFuture: il risultato arriva dopo
- Funziona tramite proxy: niente chiamate dalla stessa classe
- @EnableAsync crea il proxy

### Thread virtuali

- spring.threads.virtual.enabled: true
- Async, scheduler e Tomcat su thread virtuali
- Ideali per lavori che aspettano rete o disco
- Java 21 o successivo

### Che cosa non serve (KISS)

- Nessun pool di thread configurato a mano
- Nessun Thread.sleep nei test: Awaitility
- Più istanze: il job gira su ognuna, serve un lock condiviso
- Le eccezioni di un metodo Async finiscono nel futuro

`src/test/java/com/shop/SchedulingTests.java`

~~~java
package com.shop;

import com.shop.jobs.InventoryJobs;
import com.shop.order.Confirmation;
import com.shop.order.ConfirmationService;
import java.time.Duration;
import java.time.LocalDateTime;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.scheduling.support.CronExpression;
import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;

@SpringBootTest(properties = "shop.jobs.stock-check-delay=100ms")
class SchedulingTests {
    @Autowired ConfirmationService confirmations;
    @Autowired InventoryJobs jobs;
    @Value("${shop.jobs.cleanup-cron}") String cleanupCron;

    @Test
    void confirmationRunsOnAnotherThread() {
        Confirmation confirmation = confirmations.send(42).join();
        assertThat(confirmation.thread()).isNotEqualTo(Thread.currentThread().getName());
    }

    @Test
    void confirmationRunsOnAVirtualThread() {
        assertThat(confirmations.send(42).join().virtual()).isTrue();
    }

    @Test
    void stockCheckRunsAgainAndAgain() {
        await().atMost(Duration.ofSeconds(2)).until(() -> jobs.stockChecks() >= 2);
    }

    @Test
    void cleanupRunsAtTwoAtNight() {
        LocalDateTime noon = LocalDateTime.of(2026, 10, 16, 12, 0);
        LocalDateTime next = CronExpression.parse(cleanupCron).next(noon);
        assertThat(next).isEqualTo(LocalDateTime.of(2026, 10, 17, 2, 0));
    }
}
~~~

`src/main/java/com/shop/jobs/InventoryJobs.java`

~~~java
package com.shop.jobs;

import java.util.concurrent.atomic.AtomicInteger;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

@Component
public class InventoryJobs {
    private static final Logger LOGGER = LoggerFactory.getLogger(InventoryJobs.class);
    private final AtomicInteger stockChecks = new AtomicInteger();

    @Scheduled(fixedDelayString = "${shop.jobs.stock-check-delay}")
    public void checkStock() {
        int run = stockChecks.incrementAndGet();
        LOGGER.info("Stock check {} on {}", run, Thread.currentThread());
    }

    @Scheduled(cron = "${shop.jobs.cleanup-cron}")
    public void cleanUpExpiredCarts() {
        LOGGER.info("Cleaning up expired carts on {}", Thread.currentThread());
    }

    public int stockChecks() {
        return stockChecks.get();
    }
}
~~~

`src/main/java/com/shop/order/ConfirmationService.java`

~~~java
package com.shop.order;

import java.util.concurrent.CompletableFuture;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;

@Service
public class ConfirmationService {
    private static final Logger LOGGER = LoggerFactory.getLogger(ConfirmationService.class);

    @Async
    public CompletableFuture<Confirmation> send(long orderId) {
        Thread thread = Thread.currentThread();
        LOGGER.info("Sending confirmation for order {}", orderId);
        return CompletableFuture.completedFuture(
                new Confirmation(orderId, thread.getName(), thread.isVirtual()));
    }
}
~~~

`src/main/java/com/shop/web/ConfirmationController.java`

~~~java
package com.shop.web;

import com.shop.order.Confirmation;
import com.shop.order.ConfirmationService;
import java.util.concurrent.CompletableFuture;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class ConfirmationController {
    private final ConfirmationService confirmations;

    public ConfirmationController(ConfirmationService confirmations) {
        this.confirmations = confirmations;
    }

    @PostMapping("/api/orders/{id}/confirmation")
    public CompletableFuture<Confirmation> confirm(@PathVariable long id) {
        return confirmations.send(id);
    }
}
~~~
