# Lezione 87. Metriche e health check

## Punti chiave

### Actuator e Micrometer

- Actuator espone endpoint operativi: health, metrics, prometheus
- Via HTTP, per default, solo health
- Micrometer: API unica per le metriche, come SLF4J per i log
- Il registry sceglie il formato: Prometheus, OTLP, altri

### Liveness e readiness

- Liveness: il processo è vivo? Se no, riavviare
- Readiness: può ricevere traffico? Se no, aspettare
- Il catalogo entra solo nella readiness
- Componenti visibili, dettagli solo nel gruppo readiness

### Meter personalizzati

- Counter: valore che sale, prodotti aggiunti
- Timer: numero di chiamate e durata di list
- MeterRegistry iniettato nel costruttore
- Nomi con i punti; Prometheus: shop_products_added_total
- Boot 4: HealthIndicator in org.springframework.boot.health.contributor

### Che cosa non serve (KISS)

- Nessun asterisco: esporre solo ciò che serve
- Nessun contatore per ogni metodo: Boot misura già HTTP, JVM, thread
- Nessun dettaglio interno nella health generale
- In produzione: endpoint protetti o porta di gestione interna

`src/main/java/com/shop/catalog/ProductService.java`

~~~java
package com.shop.catalog;

import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import java.util.List;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private final ProductRepository repository;
    private final Counter addedProducts;
    private final Timer listTimer;

    public ProductService(ProductRepository repository, MeterRegistry registry) {
        this.repository = repository;
        this.addedProducts = Counter.builder("shop.products.added").register(registry);
        this.listTimer = Timer.builder("shop.products.list").register(registry);
    }

    public List<Product> list() {
        return listTimer.record(repository::findAll);
    }

    public Product find(long id) {
        return repository.findById(id).orElseThrow(() -> new ProductNotFoundException(id));
    }

    public Product create(String name, int priceInCents) {
        Product product = repository.save(new Product(0, name, priceInCents));
        addedProducts.increment();
        return product;
    }

    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

`src/main/java/com/shop/catalog/CatalogHealthIndicator.java`

~~~java
package com.shop.catalog;

import org.springframework.boot.health.contributor.Health;
import org.springframework.boot.health.contributor.HealthIndicator;
import org.springframework.stereotype.Component;

@Component
public class CatalogHealthIndicator implements HealthIndicator {
    private final ProductRepository repository;

    public CatalogHealthIndicator(ProductRepository repository) {
        this.repository = repository;
    }

    @Override
    public Health health() {
        int products = repository.findAll().size();
        if (products == 0) {
            return Health.down().withDetail("products", 0).build();
        }
        return Health.up().withDetail("products", products).build();
    }
}
~~~

`src/main/java/com/shop/catalog/InMemoryProductRepository.java`

~~~java
package com.shop.catalog;

import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;
import org.springframework.stereotype.Repository;

@Repository
public class InMemoryProductRepository implements ProductRepository {
    private final Map<Long, Product> products = new ConcurrentHashMap<>();
    private final AtomicLong sequence = new AtomicLong();

    public InMemoryProductRepository() {
        save(new Product(0, "Keyboard", 4900));
        save(new Product(0, "Mouse", 2500));
        save(new Product(0, "Monitor", 18900));
    }

    @Override
    public List<Product> findAll() {
        return products.values().stream().sorted(Comparator.comparingLong(Product::id)).toList();
    }

    @Override
    public Optional<Product> findById(long id) {
        return Optional.ofNullable(products.get(id));
    }

    @Override
    public Product save(Product product) {
        long id = product.id() == 0 ? sequence.incrementAndGet() : product.id();
        Product stored = new Product(id, product.name(), product.priceInCents());
        products.put(id, stored);
        return stored;
    }

    @Override
    public boolean deleteById(long id) {
        return products.remove(id) != null;
    }
}
~~~

`src/main/java/com/shop/web/ApiErrorHandler.java`

~~~java
package com.shop.web;

import com.shop.catalog.ProductNotFoundException;
import java.net.URI;
import java.util.Map;
import java.util.TreeMap;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

@RestControllerAdvice
public class ApiErrorHandler extends ResponseEntityExceptionHandler {
    @ExceptionHandler(ProductNotFoundException.class)
    public ProblemDetail productNotFound(ProductNotFoundException exception) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, exception.getMessage());
        problem.setType(URI.create("https://api.shop.example/problems/product-not-found"));
        problem.setTitle("Product not found");
        return problem;
    }

    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(
            MethodArgumentNotValidException exception,
            HttpHeaders headers,
            HttpStatusCode status,
            WebRequest request) {
        Map<String, String> errors = new TreeMap<>();
        exception.getBindingResult().getFieldErrors().forEach(error ->
                errors.putIfAbsent(error.getField(), error.getDefaultMessage()));
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(status, "One or more fields are invalid.");
        problem.setType(URI.create("https://api.shop.example/problems/validation-error"));
        problem.setTitle("Request validation failed");
        problem.setProperty("errors", errors);
        return handleExceptionInternal(exception, problem, headers, status, request);
    }
}
~~~

~~~java
package com.shop;

import io.micrometer.core.instrument.MeterRegistry;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;
import static org.hamcrest.Matchers.containsString;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;

@SpringBootTest
@AutoConfigureMockMvc
class ObservabilityTests {
    @Autowired MockMvc mvc;
    @Autowired MeterRegistry registry;

    @Test
    void healthShowsComponentsWithoutDetails() throws Exception {
        mvc.perform(get("/actuator/health"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.components.catalog.status").value("UP"))
                .andExpect(jsonPath("$.components.diskSpace.details").doesNotExist());
    }

    @Test
    void readinessDependsOnTheCatalog() throws Exception {
        mvc.perform(get("/actuator/health/readiness"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.components.catalog.status").value("UP"))
                .andExpect(jsonPath("$.components.catalog.details.products").isNumber());
    }

    @Test
    void livenessDoesNotDependOnTheCatalog() throws Exception {
        mvc.perform(get("/actuator/health/liveness"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.components.catalog").doesNotExist());
    }

    @Test
    void addingAProductIncrementsTheCounter() throws Exception {
        double before = registry.get("shop.products.added").counter().count();
        mvc.perform(post("/api/products").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\"Webcam\",\"priceInCents\":3900}"))
                .andExpect(status().isCreated());
        assertEquals(before + 1, registry.get("shop.products.added").counter().count());
    }

    @Test
    void prometheusExposesTheCounter() throws Exception {
        mvc.perform(get("/actuator/prometheus"))
                .andExpect(status().isOk())
                .andExpect(content().string(containsString("shop_products_added_total")));
    }
}
~~~
