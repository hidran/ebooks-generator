# Lezione 79. SecurityFilterChain e password come hash

## Punti chiave

### Password salvate come hash

- Non si cifrano: hash lento con sale (BCrypt)
- Stessa password, hash sempre diverso
- Il costo rallenta chi prova a indovinare
- DelegatingPasswordEncoder: prefisso {bcrypt}
- Mai password in chiaro nel codice o nel log

### Prima e dopo

- Prima: controllo dei ruoli nel controller
- Prima: regola da ripetere per ogni endpoint
- Dopo: controller con una sola responsabilità (SRP)
- Dopo: regole in un solo posto, leggibili (KISS)

### Le regole in ordine

- Vince la prima regola che corrisponde
- Eccezioni pubbliche in alto
- hasRole("ADMIN") per le cancellazioni
- anyRequest().authenticated() sempre in fondo
- httpBasic, CSRF disattivato, STATELESS

`src/main/java/com/shop/security/SecurityConfig.java`

~~~java
package com.shop.security;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
public class SecurityConfig {
    @Bean
    PasswordEncoder passwordEncoder() {
        return PasswordEncoderFactories.createDelegatingPasswordEncoder();
    }
    @Bean
    UserDetailsService users() {
        // Demo users of the course: passwords alice-demo and bob-demo, stored only as BCrypt hashes
        return new InMemoryUserDetailsManager(
                User.withUsername("alice")
                        .password("{bcrypt}$2a$10$YZBK.GYMthlDx9tf0GMnxOd9MBVpx8dsg0EKXSk3FkWYnLG8yd0Re")
                        .roles("USER").build(),
                User.withUsername("bob")
                        .password("{bcrypt}$2a$10$Z8sbHRaI1LujirsaEdW9SuECMKKoR8xX0auvzxjYig/099s/d4k2e")
                        .roles("ADMIN").build());
    }
    @Bean
    SecurityFilterChain api(HttpSecurity http) throws Exception {
        return http
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/rest-client.html", "/requests/**", "/error").permitAll()
                        .requestMatchers(HttpMethod.GET, "/api/products/**").permitAll()
                        .requestMatchers(HttpMethod.DELETE, "/api/products/**").hasRole("ADMIN")
                        .anyRequest().authenticated())
                .httpBasic(Customizer.withDefaults())
                .csrf(csrf -> csrf.disable())
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .build();
    }
}
~~~

`src/test/java/com/shop/security/PasswordStorageTests.java`

~~~java
package com.shop.security;

import static org.assertj.core.api.Assertions.assertThat;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.password.PasswordEncoder;

@SpringBootTest
class PasswordStorageTests {
    @Autowired
    PasswordEncoder encoder;

    @Autowired
    UserDetailsService users;

    @Test
    void sameDemoPasswordGivesDifferentSaltedHashes() {
        String first = encoder.encode("bob-demo");
        String second = encoder.encode("bob-demo");
        assertThat(first).startsWith("{bcrypt}").isNotEqualTo(second);
        assertThat(encoder.matches("bob-demo", first)).isTrue();
        assertThat(encoder.matches("bob-demo", second)).isTrue();
    }

    @Test
    void storedPasswordIsOnlyAHash() {
        String stored = users.loadUserByUsername("bob").getPassword();
        assertThat(stored).startsWith("{bcrypt}").doesNotContain("bob-demo");
        assertThat(encoder.matches("bob-demo", stored)).isTrue();
        assertThat(encoder.matches("wrong-password", stored)).isFalse();
    }
}
~~~

`src/test/java/com/shop/web/ProductSecurityTests.java`

~~~java
package com.shop.web;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.delete;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

@SpringBootTest
@AutoConfigureMockMvc
class ProductSecurityTests {
    private static final String ALICE = basic("alice", "alice-demo");
    private static final String BOB = basic("bob", "bob-demo");
    private static final String NEW_PRODUCT = "{\"name\":\"Webcam\",\"priceInCents\":5900}";

    @Autowired
    MockMvc mvc;

    @Test
    void catalogIsPublicForReading() throws Exception {
        mvc.perform(get("/api/products"))
                .andExpect(status().isOk());
    }

    @Test
    void anonymousCannotCreate() throws Exception {
        mvc.perform(post("/api/products").contentType(MediaType.APPLICATION_JSON).content(NEW_PRODUCT))
                .andExpect(status().isUnauthorized());
    }

    @Test
    void aliceCreatesAProduct() throws Exception {
        mvc.perform(post("/api/products").header(HttpHeaders.AUTHORIZATION, ALICE)
                        .contentType(MediaType.APPLICATION_JSON).content(NEW_PRODUCT))
                .andExpect(status().isCreated());
    }

    @Test
    void aliceCannotDelete() throws Exception {
        mvc.perform(delete("/api/products/2").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isForbidden());
    }

    @Test
    void bobDeletesAProduct() throws Exception {
        mvc.perform(delete("/api/products/3").header(HttpHeaders.AUTHORIZATION, BOB))
                .andExpect(status().isNoContent());
    }

    private static String basic(String username, String password) {
        String credentials = username + ":" + password;
        return "Basic " + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.UTF_8));
    }
}
~~~

## Riepilogo

- Un bean SecurityFilterChain sostituisce la catena predefinita
- DSL a lambda, dalla regola più specifica alla più generale
- Password solo come hash BCrypt
- Autorizzazione fuori dal controller
- CSRF disattivato perché non ci sono sessioni
