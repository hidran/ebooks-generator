# Lezione 86. Sicurezza delle API, OWASP e segreti

## Punti chiave

### OWASP API Top 10: i primi cinque

- API1 autorizzazione sugli oggetti (BOLA)
- API2 autenticazione debole
- API3 proprietà esposte o accettate per errore
- API4 consumo di risorse senza limiti
- API5 autorizzazione sulle funzioni

### OWASP API Top 10: gli altri cinque

- API6 flussi di business sensibili
- API7 server side request forgery
- API8 configurazione sbagliata
- API9 inventario delle API
- API10 fiducia cieca in altre API

### Segreti fuori dal codice

- Mai nel codice o nei file versionati
- Dall'ambiente o da un gestore di segreti
- Nessun valore predefinito: avvio fermato se manca
- Mai nei log, negli errori o nei Problem Detail
- Uno per ambiente, da ruotare

~~~java
package com.shop.order;

import org.springframework.stereotype.Service;

@Service
public class OrderService {
    private final OrderRepository repository;

    public OrderService(OrderRepository repository) {
        this.repository = repository;
    }

    public Order findForCustomer(long id, String username) {
        return repository.findById(id)
                .filter(order -> order.owner().equals(username))
                .orElseThrow(() -> new OrderNotFoundException(id));
    }
}
~~~

`src/main/java/com/shop/web/ApiErrorHandler.java`

~~~java
package com.shop.web;

import com.shop.catalog.ProductNotFoundException;
import com.shop.order.OrderNotFoundException;
import java.net.URI;
import java.util.Map;
import java.util.TreeMap;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

@RestControllerAdvice
public class ApiErrorHandler extends ResponseEntityExceptionHandler {
    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(MethodArgumentNotValidException exception,
            HttpHeaders headers, HttpStatusCode status, WebRequest request) {
        Map<String, String> errors = new TreeMap<>();
        exception.getBindingResult().getFieldErrors()
                .forEach(error -> errors.putIfAbsent(error.getField(), error.getDefaultMessage()));
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(status, "One or more fields are invalid.");
        problem.setType(URI.create("https://api.shop.example/problems/validation-error"));
        problem.setTitle("Request validation failed");
        problem.setProperty("errors", errors);
        return handleExceptionInternal(exception, problem, headers, status, request);
    }

    @ExceptionHandler(ProductNotFoundException.class)
    public ProblemDetail productNotFound(ProductNotFoundException exception) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, exception.getMessage());
        problem.setType(URI.create("https://api.shop.example/problems/product-not-found"));
        problem.setTitle("Product not found");
        return problem;
    }

    @ExceptionHandler(OrderNotFoundException.class)
    public ProblemDetail orderNotFound(OrderNotFoundException exception) {
        ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, exception.getMessage());
        problem.setType(URI.create("https://api.shop.example/problems/order-not-found"));
        problem.setTitle("Order not found");
        return problem;
    }
}
~~~

~~~java
package com.shop.web;

import com.shop.order.OrderService;
import java.security.Principal;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class OrderController {
    private final OrderService service;

    public OrderController(OrderService service) {
        this.service = service;
    }

    @GetMapping("/api/orders/{id}")
    public OrderResponse find(@PathVariable long id, Principal principal) {
        return OrderResponse.from(service.findForCustomer(id, principal.getName()));
    }
}
~~~

~~~java
package com.shop.order;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.HttpHeaders;
import org.springframework.test.web.servlet.MockMvc;

@SpringBootTest(properties = "shop.payment.api-key=test-only-value")
@AutoConfigureMockMvc
class OrderAccessTests {
    private static final String ALICE = basic("alice", "alice-demo");
    private static final String NOT_FOUND = "https://api.shop.example/problems/order-not-found";

    @Autowired
    MockMvc mvc;

    @Test
    void aliceReadsHerOrder() throws Exception {
        mvc.perform(get("/api/orders/1").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.product").value("Keyboard"));
    }

    @Test
    void aliceCannotReadBobsOrder() throws Exception {
        mvc.perform(get("/api/orders/2").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isNotFound())
                .andExpect(jsonPath("$.type").value(NOT_FOUND));
    }

    @Test
    void missingOrderLooksExactlyTheSame() throws Exception {
        mvc.perform(get("/api/orders/99").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isNotFound())
                .andExpect(jsonPath("$.type").value(NOT_FOUND));
    }

    @Test
    void anonymousCannotReadOrders() throws Exception {
        mvc.perform(get("/api/orders/1"))
                .andExpect(status().isUnauthorized());
    }

    private static String basic(String username, String password) {
        String credentials = username + ":" + password;
        return "Basic " + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.UTF_8));
    }
}
~~~
