# Lezione 83. Ruoli e method security

## Punti chiave

### Regole sugli indirizzi o sui metodi

- La catena protegge indirizzi HTTP
- Pre Authorize protegge un'operazione
- @EnableMethodSecurity attiva il proxy
- Il proxy valuta l'espressione prima del metodo

> Percorso: Richiesta, poi Catena (URL), poi Controller, poi Proxy (@PreAuthorize), poi ProductService.

### Prima e dopo

- Prima: regola DELETE legata a /api/products
- Prima: /api/clearance la aggira
- Dopo: @PreAuthorize("hasRole('ADMIN')") su delete
- Una sola fonte di verità (DRY), espressione corta (KISS)

### Regole chiare

- Annotazioni sui servizi, non sui controller
- hasRole, hasAnyRole: espressioni brevi
- Le chiamate interne alla stessa classe non passano dal proxy
- Regole generali nella catena, operazioni sui metodi

~~~java
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;

@RestController
public class ClearanceController {

    public ClearanceController(ProductService service) {
        this.service = service;
    }

    // Added later by another team: removes a product sold out in a clearance sale
    @PostMapping("/api/clearance/{id}")

        service.delete(id);
        return ResponseEntity.noContent().build();
    }
}
~~~

`src/main/java/com/shop/catalog/ProductService.java`

~~~java
package com.shop.catalog;

import java.util.List;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private final ProductRepository repository;

    public ProductService(ProductRepository repository) {
        this.repository = repository;
    }

    public List<Product> list() {
        return repository.findAll();
    }

    public Product find(long id) {
        return repository.findById(id).orElseThrow(() -> new ProductNotFoundException(id));
    }

    public Product create(String name, int priceInCents) {
        return repository.save(new Product(0, name, priceInCents));
    }

    @PreAuthorize("hasRole('ADMIN')")
    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

~~~java
package com.shop.security;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.method.configuration.EnableMethodSecurity;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.factory.PasswordEncoderFactories;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableMethodSecurity
public class SecurityConfig {
    @Bean
    PasswordEncoder passwordEncoder() {
        return PasswordEncoderFactories.createDelegatingPasswordEncoder();
    }
    @Bean
    UserDetailsService users() {
        // Demo users of the course: passwords alice-demo and bob-demo, stored only as BCrypt hashes
        return new InMemoryUserDetailsManager(
                User.withUsername("alice")
                        .password("{bcrypt}$2a$10$YZBK.GYMthlDx9tf0GMnxOd9MBVpx8dsg0EKXSk3FkWYnLG8yd0Re")
                        .roles("USER").build(),
                User.withUsername("bob")
                        .password("{bcrypt}$2a$10$Z8sbHRaI1LujirsaEdW9SuECMKKoR8xX0auvzxjYig/099s/d4k2e")
                        .roles("ADMIN").build());
    }
    @Bean
    SecurityFilterChain api(HttpSecurity http) throws Exception {
        return http
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/rest-client.html", "/requests/**", "/error").permitAll()
                        .requestMatchers(HttpMethod.GET, "/api/products/**").permitAll()
                        .anyRequest().authenticated())
                .httpBasic(Customizer.withDefaults())
                .csrf(csrf -> csrf.disable())
                .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .build();
    }
}
~~~

~~~java
package com.shop.web;

import static org.assertj.core.api.Assertions.assertThatThrownBy;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.delete;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

import com.shop.catalog.ProductService;
import java.nio.charset.StandardCharsets;
import java.util.Base64;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.HttpHeaders;
import org.springframework.security.authentication.AuthenticationCredentialsNotFoundException;
import org.springframework.test.web.servlet.MockMvc;

@SpringBootTest
@AutoConfigureMockMvc
class DeleteRulesTests {
    private static final String ALICE = basic("alice", "alice-demo");
    private static final String BOB = basic("bob", "bob-demo");

    @Autowired
    MockMvc mvc;

    @Autowired
    ProductService service;

    @Test
    void aliceCannotDeleteThroughTheProductsApi() throws Exception {
        mvc.perform(delete("/api/products/1").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isForbidden());
    }

    @Test
    void aliceCannotRemoveThroughTheClearanceEndpoint() throws Exception {
        mvc.perform(post("/api/clearance/1").header(HttpHeaders.AUTHORIZATION, ALICE))
                .andExpect(status().isForbidden());
    }

    @Test
    void bobDeletesThroughTheProductsApi() throws Exception {
        mvc.perform(delete("/api/products/2").header(HttpHeaders.AUTHORIZATION, BOB))
                .andExpect(status().isNoContent());
    }

    @Test
    void bobRemovesThroughTheClearanceEndpoint() throws Exception {
        mvc.perform(post("/api/clearance/3").header(HttpHeaders.AUTHORIZATION, BOB))
                .andExpect(status().isNoContent());
    }

    @Test
    void serviceRefusesACallerWithoutAuthentication() {
        assertThatThrownBy(() -> service.delete(1))
                .isInstanceOf(AuthenticationCredentialsNotFoundException.class);
    }

    private static String basic(String username, String password) {
        String credentials = username + ":" + password;
        return "Basic " + Base64.getEncoder().encodeToString(credentials.getBytes(StandardCharsets.UTF_8));
    }
}
~~~
