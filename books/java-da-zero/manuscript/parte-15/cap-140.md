# Lezione 114. TypeScript per chi sa Java

## Punti chiave

### JavaScript con i tipi

- TypeScript: JavaScript più i tipi
- L'editor e il compilatore controllano i tipi
- Durante l'esecuzione i tipi spariscono
- Conta la forma dell'oggetto, non il nome della classe
- Angular è scritto in TypeScript

### Node esegue i file .ts

- Node: JavaScript fuori dal browser, serve anche ad Angular
- Da Node 22.18 o 24.12 in poi: node shop.ts e basta
- Toglie i tipi senza controllarli: il controllo è dell'editor
- Niente enum né namespace con codice
- Niente compilatore, niente pacchetti da installare

> Percorso: shop.ts, poi Node toglie i tipi, poi JavaScript, poi Output.

package com.shop.web;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import java.util.List;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

@Tag(name = "Login", description = "Exchange username and password for a token")
@RestController
public class MeController {
    @Operation(summary = "Who am I: the user and the roles of the token")
    @GetMapping("/api/me")
    public MeResponse me(Authentication authentication) {
        List<String> roles = authentication.getAuthorities().stream()
                .map(GrantedAuthority::getAuthority)
                .filter(authority -> authority.startsWith("ROLE_"))
                .map(authority -> authority.substring("ROLE_".length()))
                .toList();
        return new MeResponse(authentication.getName(), roles);
    }
}

package com.shop;

import static org.assertj.core.api.Assertions.assertThat;

import com.jayway.jsonpath.JsonPath;
import com.shop.web.OrderResponse;
import com.shop.web.ProductResponse;
import java.util.List;
import java.util.Map;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.resttestclient.autoconfigure.AutoConfigureRestTestClient;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.context.SpringBootTest.WebEnvironment;
import org.springframework.http.MediaType;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.web.servlet.client.RestTestClient;

// The whole application on a random port, called over real HTTP, as the browser does.
@SpringBootTest(webEnvironment = WebEnvironment.RANDOM_PORT)
@AutoConfigureRestTestClient
@ActiveProfiles("test")
class ShopEndToEndTests {
    @Autowired
    RestTestClient client;

    @Test
    void guestFindsAProductAndOrdersIt() {
        client.get().uri("/api/products?name=mouse&sort=PRICE")
                .exchange()
                .expectStatus().isOk()
                .expectBody()
                .jsonPath("$.totalElements").isEqualTo(2)
                .jsonPath("$.content[0].name").isEqualTo("Mouse pad");
        client.get().uri("/api/products/2")
                .exchange()
                .expectStatus().isOk()
                .expectBody().jsonPath("$.categoryName").isEqualTo("Peripherals");
        OrderResponse order = client.post().uri("/api/orders")
                .contentType(MediaType.APPLICATION_JSON)
                .body(Map.of("lines", List.of(Map.of("productId", 2, "quantity", 3))))
                .exchange()
                .expectStatus().isCreated()
                .expectBody(OrderResponse.class).returnResult().getResponseBody();
        assertThat(order.totalInCents()).isEqualTo(7500);
    }

    @Test
    void adminLogsInAndAddsAProduct() {
        String json = client.post().uri("/api/auth/token")
                .headers(headers -> headers.setBasicAuth("admin", "admin-demo"))
                .exchange()
                .expectStatus().isOk()
                .expectBody(String.class).returnResult().getResponseBody();
        String token = JsonPath.read(json, "$.token");
        client.get().uri("/api/me")
                .headers(headers -> headers.setBearerAuth(token))
                .exchange()
                .expectStatus().isOk()
                .expectBody().jsonPath("$.roles[0]").isEqualTo("ADMIN");
        ProductResponse created = client.post().uri("/api/admin/products")
                .headers(headers -> headers.setBearerAuth(token))
                .contentType(MediaType.APPLICATION_JSON)
                .body(Map.of("name", "Webcam 4K", "description", "Sharp webcam.", "priceInCents", 9900, "categoryId", 1))
                .exchange()
                .expectStatus().isCreated()
                .expectBody(ProductResponse.class).returnResult().getResponseBody();
        client.get().uri("/api/products/" + created.id())
                .exchange()
                .expectStatus().isOk()
                .expectBody().jsonPath("$.name").isEqualTo("Webcam 4K");
    }

    @Test
    void customerCannotChangeTheCatalog() {
        String json = client.post().uri("/api/auth/token")
                .headers(headers -> headers.setBasicAuth("customer", "customer-demo"))
                .exchange()
                .expectStatus().isOk()
                .expectBody(String.class).returnResult().getResponseBody();
        String token = JsonPath.read(json, "$.token");
        client.post().uri("/api/admin/products")
                .headers(headers -> headers.setBearerAuth(token))
                .contentType(MediaType.APPLICATION_JSON)
                .body(Map.of("name", "Cheap keyboard", "description", "Cheaper.", "priceInCents", 100, "categoryId", 1))
                .exchange()
                .expectStatus().isForbidden();
    }

    @Test
    void unknownProductIsAProblemDetail() {
        client.get().uri("/api/products/999")
                .exchange()
                .expectStatus().isNotFound()
                .expectHeader().contentType(MediaType.APPLICATION_PROBLEM_JSON)
                .expectBody().jsonPath("$.title").isEqualTo("Product not found");
    }
}

package com.shop.security;

import static org.springframework.security.test.web.servlet.request.SecurityMockMvcRequestPostProcessors.httpBasic;
import static org.springframework.security.test.web.servlet.request.SecurityMockMvcRequestPostProcessors.jwt;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.options;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.header;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

import com.jayway.jsonpath.JsonPath;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.HttpHeaders;
import org.springframework.http.MediaType;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.web.servlet.MockMvc;
import org.springframework.transaction.annotation.Transactional;

@SpringBootTest
@AutoConfigureMockMvc
@ActiveProfiles("test")
@Transactional
class SecurityRulesTests {
    private static final String DEV_SERVER = "http://127.0.0.1:21141";
    private static final String NEW_PRODUCT = """
            {"name":"Webcam HD","description":"Full HD webcam.","priceInCents":6900,"categoryId":1}
            """;

    @Autowired
    MockMvc mvc;

    @Test
    void anonymousReadsCatalogAndCategories() throws Exception {
        mvc.perform(get("/api/products")).andExpect(status().isOk());
        mvc.perform(get("/api/products/1")).andExpect(status().isOk());
        mvc.perform(get("/api/categories")).andExpect(status().isOk());
    }

    @Test
    void guestPlacesAnOrder() throws Exception {
        mvc.perform(post("/api/orders").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"lines\":[{\"productId\":3,\"quantity\":1}]}"))
                .andExpect(status().isCreated())
                .andExpect(jsonPath("$.totalInCents").value(18900));
    }

    @Test
    void tokenIsIssuedForTheDemoAdmin() throws Exception {
        mvc.perform(post("/api/auth/token").with(httpBasic("admin", "admin-demo")))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.token").isString())
                .andExpect(jsonPath("$.expiresInSeconds").value(900));
    }

    @Test
    void wrongPasswordFromJavaScriptGetsNoBrowserLoginDialog() throws Exception {
        mvc.perform(post("/api/auth/token").with(httpBasic("admin", "wrong"))
                        .header("X-Requested-With", "XMLHttpRequest"))
                .andExpect(status().isUnauthorized())
                .andExpect(header().doesNotExist(HttpHeaders.WWW_AUTHENTICATE));
    }

    @Test
    void meAnswersWithTheOwnerOfTheToken() throws Exception {
        String authorization = bearer("customer", "customer-demo");
        mvc.perform(get("/api/me").header(HttpHeaders.AUTHORIZATION, authorization))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.username").value("customer"))
                .andExpect(jsonPath("$.roles[0]").value("CUSTOMER"));
    }

    @Test
    void meNeedsAToken() throws Exception {
        mvc.perform(get("/api/me")).andExpect(status().isUnauthorized());
    }

    @Test
    void anonymousCannotChangeTheCatalog() throws Exception {
        mvc.perform(post("/api/admin/products").contentType(MediaType.APPLICATION_JSON).content(NEW_PRODUCT))
                .andExpect(status().isUnauthorized());
    }

    @Test
    void customerCannotChangeTheCatalog() throws Exception {
        mvc.perform(post("/api/admin/products").contentType(MediaType.APPLICATION_JSON).content(NEW_PRODUCT)
                        .with(jwt().authorities(new SimpleGrantedAuthority("ROLE_CUSTOMER"))))
                .andExpect(status().isForbidden());
    }

    @Test
    void adminTokenCreatesAProduct() throws Exception {
        mvc.perform(post("/api/admin/products").contentType(MediaType.APPLICATION_JSON).content(NEW_PRODUCT)
                        .header(HttpHeaders.AUTHORIZATION, bearer("admin", "admin-demo")))
                .andExpect(status().isCreated())
                .andExpect(header().exists(HttpHeaders.LOCATION));
    }

    @Test
    void corsAllowsTheDevServer() throws Exception {
        mvc.perform(get("/api/products").header(HttpHeaders.ORIGIN, DEV_SERVER))
                .andExpect(status().isOk())
                .andExpect(header().string(HttpHeaders.ACCESS_CONTROL_ALLOW_ORIGIN, DEV_SERVER));
    }

    @Test
    void corsRejectsAnotherOrigin() throws Exception {
        mvc.perform(get("/api/products").header(HttpHeaders.ORIGIN, "http://127.0.0.1:21999"))
                .andExpect(status().isForbidden());
    }

    @Test
    void preflightForTheLoginIsAnswered() throws Exception {
        mvc.perform(options("/api/auth/token").header(HttpHeaders.ORIGIN, DEV_SERVER)
                        .header(HttpHeaders.ACCESS_CONTROL_REQUEST_METHOD, "POST")
                        .header(HttpHeaders.ACCESS_CONTROL_REQUEST_HEADERS, "authorization,x-requested-with"))
                .andExpect(status().isOk())
                .andExpect(header().string(HttpHeaders.ACCESS_CONTROL_ALLOW_ORIGIN, DEV_SERVER));
    }

    private String bearer(String username, String password) throws Exception {
        String json = mvc.perform(post("/api/auth/token").with(httpBasic(username, password)))
                .andExpect(status().isOk())
                .andReturn().getResponse().getContentAsString();
        return "Bearer " + JsonPath.read(json, "$.token");
    }
}

## Riepilogo

- Optional: ?. e ??; lambda: =&gt;
- Template literal, spread e destructuring
- CompletableFuture: Promise; join: await, senza bloccare
