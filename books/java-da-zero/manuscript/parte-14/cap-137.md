# Lezione 111. Revisione finale e checklist

## Punti chiave

### SOLID, DRY e KISS nello Shop

- S: controller, servizio e repository, un compito ciascuno
- O e D: ProductService dipende dalla porta ProductRepository
- Dalla memoria a JPA senza toccare il servizio
- DRY: ogni regola vive in un posto solo
- KISS: niente astrazioni che oggi non servono

### Checklist: il codice

- Iniezione nel costruttore, campi final
- Strati separati, DTO distinti dal dominio
- Errori ProblemDetail, validazione al confine
- Piramide dei test: unità, slice, pochi test completi

### Checklist: la produzione

- Configurazione dall'ambiente, dodici fattori
- Schema versionato con le migrazioni di Flyway
- Log strutturati, metriche e health check
- Sicurezza: ruoli chiari e token validati
- Immagini non root, segreti fuori dal codice

### Prima e dopo

- Prima: @Autowired sul campo e ProductRepository nel controller
- Prima: il controller ripete il calcolo del servizio
- Dopo: costruttore, campo final, delega a ProductService
- Principi: SRP, DRY, inversione delle dipendenze

> Percorso: ProductStatsController, poi ProductService, poi ProductRepository.

### Il nostro percorso

- Dal primo programma Java a Spring Boot 4
- REST, persistenza, sicurezza e osservabilità
- Container, microservizi, pipeline e immagine nativa
- Uno Shop cresciuto un passo alla volta

`src/main/java/com/shop/catalog/ProductService.java`

~~~java
package com.shop.catalog;

import java.util.List;
import org.springframework.stereotype.Service;

@Service
public class ProductService {
    private final ProductRepository repository;

    public ProductService(ProductRepository repository) {
        this.repository = repository;
    }

    public List<Product> list() {
        return repository.findAll();
    }

    public Product find(long id) {
        return repository.findById(id).orElseThrow(() -> new ProductNotFoundException(id));
    }

    public Product create(String name, int priceInCents) {
        return repository.save(new Product(0, name, priceInCents));
    }

    public CatalogStats stats() {
        List<Product> products = repository.findAll();
        long total = products.stream().mapToLong(Product::priceInCents).sum();
        return new CatalogStats(products.size(), total);
    }

    public void delete(long id) {
        if (!repository.deleteById(id)) {
            throw new ProductNotFoundException(id);
        }
    }
}
~~~

~~~java
package com.shop.catalog;

import org.junit.jupiter.api.Test;
import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class ProductServiceTests {
    private final ProductService service = new ProductService(new InMemoryProductRepository());

    @Test
    void createAssignsTheNextId() {
        Product product = service.create("Webcam", 5900);
        assertThat(product.id()).isEqualTo(4);
        assertThat(service.find(4)).isEqualTo(product);
    }

    @Test
    void unknownProductIsReported() {
        assertThatThrownBy(() -> service.find(99))
                .isInstanceOf(ProductNotFoundException.class)
                .hasMessage("Product 99 does not exist.");
    }

    @Test
    void statsSumTheWholeCatalog() {
        assertThat(service.stats()).isEqualTo(new CatalogStats(3, 26300));
    }
}
~~~

`src/test/java/com/shop/web/ProductApiTests.java`

~~~java
package com.shop.web;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.webmvc.test.autoconfigure.AutoConfigureMockMvc;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class ProductApiTests {
    @Autowired
    MockMvc mvc;

    @Test
    void listReturnsTheDemoCatalog() throws Exception {
        mvc.perform(get("/api/products"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$[0].name").value("Keyboard"));
    }

    @Test
    void unknownProductIsAProblemDetail() throws Exception {
        mvc.perform(get("/api/products/99"))
                .andExpect(status().isNotFound())
                .andExpect(jsonPath("$.type").value("https://api.shop.example/problems/product-not-found"));
    }

    @Test
    void invalidProductIsAValidationProblem() throws Exception {
        mvc.perform(post("/api/products").contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\" \",\"priceInCents\":0}"))
                .andExpect(status().isBadRequest())
                .andExpect(jsonPath("$.type").value("https://api.shop.example/problems/validation-error"))
                .andExpect(jsonPath("$.errors.name").exists())
                .andExpect(jsonPath("$.errors.priceInCents").exists());
    }

    @Test
    void statsSummarizeTheCatalog() throws Exception {
        mvc.perform(get("/api/products/stats"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.productCount").value(3))
                .andExpect(jsonPath("$.totalPriceInCents").value(26300));
    }
}
~~~
